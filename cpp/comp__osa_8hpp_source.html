1/**
2 * @file comp_osa.hpp
3 * @brief One-step-ahead (OSA) residual helpers for composition data.
4 *
5 * TMB's oneStepPredict() requires every observation entering the likelihood to
6 * live in a flat `obsvec` with a `keep` indicator, and each likelihood term to
7 * be multiplied by its keep element. Multinomial and Dirichlet-multinomial
8 * composition data are multivariate and discrete, so oneStepPredict cannot use
9 * them directly. Following Trijoulet et al. (2023) and WHAM
10 * (src/age_comp_osa.hpp), a composition of A bins is decomposed into A-1
11 * univariate conditional distributions -- binomials for the multinomial,
12 * beta-binomials for the Dirichlet-multinomial -- each gated by its keep
13 * element. The final bin is fixed by the sum-to-N constraint and contributes
14 * nothing (its residual is not defined and is reported as NA on the R side).
15 *
16 * Key property: with keep == 1 the sum of the conditional log-densities equals
17 * the joint multinomial / Dirichlet-multinomial log-density. Turning OSA on
18 * therefore changes how the likelihood is *decomposed*, not its value.
19 *
20 * These functions are named with an `_osa` suffix and take a data_indicator so
21 * they never collide with TMB's built-in dmultinom() or the plain
22 * ddirmultinom() in helper_functions.hpp used during ordinary fitting.
23 *
24 * The file also holds the conditional CDFs oneStepPredict(method = "cdf") asks
25 * for -- the continuous binomial for a composition bin, and the normal /
26 * left-truncated normal for the aggregate series -- gated by
27 * keep.cdf_lower / keep.cdf_upper. Under that method the residual is
28 * Phi^-1(F(x)) and no conditional mean is ever formed, so nothing can leave the
29 * support of a bounded observation. They are computed only when the caller asks
30 * (`do_cdf`, driven by osa_mode == 2), so ordinary fitting and the Gaussian OSA
31 * methods pay nothing for them.
32 */

34// Reorder categories so kept (keep > 0.5) entries come first, preserving the
35// original order within the kept and dropped groups.
36template<class Type>
37vector<int> osa_order(vector<Type> k){
38 int n = k.size();
39 vector<int> ord(n);
40 ord.setZero();
41 int at = -1;
42 for(int i = 0; i < n; ++i){
43 if(k(i) > 0.5){ ord(++at) = i; }
44 }
45 at = n;
46 for(int i = n - 1; i >= 0; --i){
47 if(k(i) < 0.5){ ord(--at) = i; }
48 }
49 return ord;
50}

52// Map the closed [0, 1] interval onto the open interval for numerical stability.
53// Identical to TMB's convenience.hpp squeeze() (eps = machine epsilon), which is
54// what WHAM uses in its OSA composition likelihood, so the two implementations
55// agree to the last digit.
56template<class Type>
57Type osa_squeeze(Type u){
58 Type eps = std::numeric_limits<double>::epsilon();
59 return (Type(1.0) - eps) * (u - Type(0.5)) + Type(0.5);
60}

62/**
63 * @brief Map a CDF onto the open interval before its logarithm is taken.
64 *
65 * Four machine epsilons wide, where osa_squeeze() uses one, and the extra width
66 * is load-bearing. oneStepPredict recovers F from 1/(1 + exp(nlcdf.lower -
67 * nlcdf.upper)) in double precision; at one epsilon that denominator is
68 * 1 + 1.11e-16, which is exactly the round-half-to-even tie above 1, so it lands
69 * on 1 or not depending on the last bits of an objective summed over thousands
70 * of observations -- and a denominator of 1 gives F = 1 and an infinite
71 * residual. Measured on BS2017SS: 6 of 4538 composition residuals came back
72 * infinite at one epsilon and none at four.
73 *
74 * The cost is where |residual| saturates: 8.04 here against 8.13 at one epsilon
75 * and 8.21 for a CDF read straight off. That ceiling is a property of double
76 * precision, not of the model -- an observation that far into the tail of its
77 * own conditional is beyond what any Q-Q plot resolves.
78 *
79 * The width is load-bearing a second way: this maps, it does not clamp, so a
80 * caller that forms F by subtracting two CDFs relies on it to absorb the
81 * cancellation. osa_trunc_norm_cdf_terms() below is the case -- its numerator
82 * can round to about -2e-16 at an observation near zero, which this shift
83 * carries back above 0 before the log. Narrowing it would return NaN there.
84 */
85template<class Type>
86Type osa_squeeze_cdf(Type u){
87 Type eps = Type(4.0) * std::numeric_limits<double>::epsilon();
88 return (Type(1.0) - eps) * (u - Type(0.5)) + Type(0.5);
89}

91/**
92 * @brief Conditional binomial CDF P(X <= x), X ~ Binomial(n, p), at fractional x.
93 *
94 * Written through the regularized incomplete beta, 1 - I_p(x + 1, n - x). That
95 * equals the binomial CDF at whole counts and stays defined and smooth at the
96 * fractional ones composition data carry: a bin count is proportion * effective
97 * sample size, so it is only an integer by coincidence. Same expression as the
98 * Trijoulet et al. (2023) reference implementation TMB ships
99 * (contrib/OSA_multivariate_dists-main/distr.hpp, `dists::pbinom`), so the two
100 * agree term for term.
101 *
102 * @param x Count in this bin.
103 * @param n Count still unplaced across this bin and every later one.
104 * @param p Probability of landing in this bin rather than a later one.
105 */
106template<class Type>
107Type osa_pbinom(Type x, Type n, Type p){
108 // n - x is what is left for the later bins. comp_offset adds a positive
109 // amount to every bin, so it is strictly positive on every path that reaches
110 // here; the floor is defensive, because pbeta at a zero shape returns NaN
111 // rather than the limit, and a NaN times a zero cdf gate is still a NaN.
112 Type nx = CppAD::CondExpLt(n - x, Type(1e-12), Type(1e-12), n - x);
113 return Type(1.0) - pbeta(p, x + Type(1.0), nx);
114}

116/**
117 * @brief One-step-ahead CDF terms for a normal observation.
118 *
119 * Every aggregate series in CEATTLE is normal on the scale `obsvec` stores it in
120 * -- log for lognormal catch and index, natural scale for a "Normal" index, and
121 * the whitened L^-1 scale for "MVN" -- so its conditional CDF is `pnorm` of the
122 * standardized residual. Returns the term to ADD to the log-likelihood; it is 0
123 * whenever the cdf gates are 0, which is every evaluation except the two
124 * oneStepPredict(method = "cdf") makes per observation.
125 *
126 * @param x Observation, on the scale it is stored in.
127 * @param mu Predicted value on that same scale.
128 * @param sd Standard deviation on that same scale.
129 * @param cdf_lower,cdf_upper The observation's `keep.cdf_lower`/`keep.cdf_upper`.
130 */
131template<class Type>
132Type osa_norm_cdf_terms(Type x, Type mu, Type sd, Type cdf_lower, Type cdf_upper){
133 Type F = osa_squeeze_cdf(pnorm((x - mu) / sd));
134 return cdf_lower * log(F) + cdf_upper * log(Type(1.0) - F);
135}

137/**
138 * @brief One-step-ahead CDF terms for a normal left-truncated at zero.
139 *
140 * With z = (x - mu)/sd and m = mu/sd, the truncated CDF is
141 * F(x) = [Phi(z) - Phi(-m)] / Phi(m) and its complement is 1 - F(x) =
142 * Phi(-z) / Phi(m), because Phi(m) + Phi(-m) = 1. Both are written in that form
143 * rather than as 1 - F, so the upper tail keeps its precision.
144 *
145 * This is exact wherever the truncation carries mass, which is what the Gaussian
146 * OSA methods cannot see: the truncation enters the density only through
147 * log Phi(m), a function of the prediction and not of the observation.
148 *
149 * @param x Observation (natural scale, > 0).
150 * @param mu Predicted index (natural scale).
151 * @param sd Observation standard deviation (absolute, natural scale).
152 * @param cdf_lower,cdf_upper The observation's `keep.cdf_lower`/`keep.cdf_upper`.
153 */
154template<class Type>
155Type osa_trunc_norm_cdf_terms(Type x, Type mu, Type sd,
156 Type cdf_lower, Type cdf_upper){
157 Type m = mu / sd;
158 Type Zm = pnorm(m);
159 Type F = osa_squeeze_cdf((pnorm((x - mu) / sd) - pnorm(-m)) / Zm);
160 Type G = osa_squeeze_cdf(pnorm(-(x - mu) / sd) / Zm);
161 return cdf_lower * log(F) + cdf_upper * log(G);
162}

164/**
165 * @brief Keep-aware multinomial log-density via conditional binomials.
166 *
167 * With `do_osa`, decomposes the A-bin multinomial into A-1 conditional binomials
168 * (each gated by its keep element) so oneStepPredict can consume it; the sum
169 * equals the joint multinomial log-density when all keeps are 1.
170 *
171 * @param x Observed counts (length A; may be non-integer, e.g. Neff * prop).
172 * @param p Predicted proportions (length A; normalized internally).
173 * @param keep Data indicator for these A bins (from keep.segment(...)).
174 * @param do_osa 1 = conditional-binomial decomposition; 0 = plain multinomial.
175 * @param give_log 1 = return the log-density.
176 * @param do_cdf 1 = also add the keep.cdf_lower / keep.cdf_upper terms that
177 * oneStepPredict(method = "cdf") reads (osa_mode == 2). 0 skips the `pbeta`
178 * entirely, so ordinary fitting and the Gaussian OSA methods are unchanged.
179 */
180template <class Type>
181Type dmultinom_osa(vector<Type> x, vector<Type> p,
182 data_indicator<vector<Type>, Type> keep,
183 int do_osa, int give_log, int do_cdf = 0)
184{
185 Type logres = 0;
186 // Normalize predicted proportions. Guard the denominator so a length/age bin
187 // with all-zero predicted mass (sum(p) == 0, e.g. a growth-transition row that
188 // is exactly zero in the tails) yields p_x == 0 rather than 0/0 = NaN; the
189 // conditional binomials below then read each squeezed-to-eps probability,
190 // matching WHAM's behaviour for vanishing predicted proportions.
191 vector<Type> p_x = p / (sum(p) + Type(1e-300));
192
193 if(do_osa){
194 vector<Type> k = keep;
195 // The cdf gates are separate members of the data_indicator, so they have to
196 // be reordered by hand alongside the keep vector. keep.segment() at the call
197 // site already carries them (tmb_core.hpp documents that its segment IS
198 // applied to cdf_lower/cdf_upper); the reorder here is the step that is not
199 // done for us, and getting it wrong gates the wrong bin silently.
200 //
201 // osa_order() branches on keep, and under oneStepPredict keep is a PARAMETER,
202 // so that branch is taped once at the initial values -- 1 for every bin that
203 // will ever be residualized or conditioned on, 0 for the rest -- and the
204 // per-observation keep changes afterwards do not re-sort. Within a
205 // composition that initial pattern is 1 on every bin but the last, so the
206 // order is the identity and each bin stays where the conditional
207 // decomposition needs it. The reorder below is therefore a no-op on every
208 // sequence this package generates; it is kept because it is what makes the
209 // decomposition right for an arbitrary keep, and it is pinned by a test that
210 // supplies one directly (test-likelihood-osa-cdf.R).
211 vector<Type> cl = keep.cdf_lower;
212 vector<Type> cu = keep.cdf_upper;
213 vector<int> o = osa_order(k);
214 x = x(o); p_x = p_x(o); k = k(o); // kept bins first
215 if(do_cdf){ cl = cl(o); cu = cu(o); }
216 Type nUnused = asDouble(sum(x));
217 Type pUsed = 0;
218 Type cdf = 1;
219 for(int i = 0; i < x.size(); ++i){
220 if(i != (x.size() - 1)){
221 // Bin i versus all remaining bins: a binomial written as a 2-category
222 // multinomial. Conditional success probability p_i / (1 - pUsed).
223 vector<Type> x2(2), p2(2);
224 x2(0) = x(i);
225 x2(1) = nUnused - x(i);
226 p2(0) = osa_squeeze(p_x(i)) / osa_squeeze(Type(1) - pUsed);
227 p2(0) = osa_squeeze(p2(0));
228 // When one bin holds ~all remaining mass the separately-squeezed
229 // numerator/denominator can push the ratio marginally above 1; clamp so
230 // p2(1) = 1 - p2(0) stays non-negative (avoids dmultinom log(negative) =
231 // NaN). A no-op for well-behaved compositions.
232 Type eps_clamp = std::numeric_limits<double>::epsilon();
233 p2(0) = CppAD::CondExpGt(p2(0), Type(1) - eps_clamp, Type(1) - eps_clamp, p2(0));
234 p2(1) = Type(1) - p2(0);
235 logres += k(i) * dmultinom(x2, p2, true);
236 // Same conditional binomial the density above scores, read as a CDF.
237 // Taken before nUnused is decremented, so the size matches the density.
238 if(do_cdf) cdf = osa_pbinom(x(i), nUnused, p2(0));
239 nUnused -= x(i);
240 pUsed += p_x(i);
241 // Keep pUsed inside (0, 1) so 1 - pUsed never reaches 0 or goes
242 // negative. WHAM does NOT squeeze pUsed itself; it relies on the
243 // squeeze of 1 - pUsed above, which is also applied here. This extra
244 // squeeze is therefore redundant protection and contracts pUsed toward
245 // 0.5 by one machine epsilon per bin, so a composition of A bins sits
246 // O(A * eps) from WHAM's value -- ~1e-15 relative, far below the
247 // reference tolerances, but not bit-identical. Removing it would match
248 // WHAM exactly at the cost of re-pinning the OSA references.
249 pUsed = osa_squeeze(pUsed);
250 } else {
251 logres += k(i) * Type(0); // last bin: fixed by sum-to-N
252 cdf = Type(1); // ... so it sits at the top of its own CDF
253 }
254 if(do_cdf){
255 // Onto the open interval before taking logs: an exactly 0 or 1 CDF puts
256 // a -Inf on the tape, and a -Inf times a zero gate is a NaN. See
257 // osa_squeeze_cdf() for why the width matters and what it caps.
258 cdf = osa_squeeze_cdf(cdf);
259 logres += cl(i) * log(cdf);
260 logres += cu(i) * log(Type(1) - cdf);
261 }
262 }
263 } else {
264 logres = dmultinom(x, p_x, true);
265 }
266 return give_log ? logres : exp(logres);
267}

269/**
270 * @brief Keep-aware Dirichlet-multinomial log-density via conditional beta-binomials.
271 *
272 * With `do_osa`, decomposes the A-bin Dirichlet-multinomial into A-1 conditional
273 * beta-binomials (each gated by its keep element); the sum equals the joint D-M
274 * log-density when all keeps are 1.
275 *
276 * There is no `do_cdf` counterpart here, so a Dirichlet-multinomial fleet cannot
277 * be residualized with oneStepPredict(method = "cdf") and osa_residuals() sends
278 * it to a Gaussian method instead. A beta-binomial has no elementary CDF, and
279 * the reference implementation's route -- summing the pmf over 0..floor(x)
280 * (contrib/OSA_multivariate_dists-main/distr.hpp, `pbetabinom`) -- is not open
281 * to us on two counts: it is a step function of a fractional count, and it costs
282 * O(x) beta functions per bin, which at CEATTLE sample sizes is thousands of
283 * times the multinomial path.
284 *
285 * @param obs Observed counts (length A).
286 * @param alpha Dirichlet-multinomial concentration parameters (length A).
287 * @param keep Data indicator for these A bins.
288 * @param do_osa 1 = conditional beta-binomial decomposition; 0 = plain D-M.
289 * @param give_log 1 = return the log-density.
290 */
291template<class Type>
292Type ddirmultinom_osa(vector<Type> obs, vector<Type> alpha,
293 data_indicator<vector<Type>, Type> keep,
294 int do_osa, int give_log)
295{
296 Type ll = 0.0;
297 if(do_osa){
298 vector<Type> k = keep;
299 vector<int> o = osa_order(k);
300 obs = obs(o); alpha = alpha(o); k = k(o); // kept bins first
301 int dim = obs.size();
302 Type alp_sum = alpha.sum();
303 Type obs_sum = asDouble(obs.sum());
304 for(int a = 0; a < dim - 1; ++a){
305 // Bin a versus all remaining bins: a beta-binomial written as a
306 // 2-category Dirichlet-multinomial (alpha_a vs sum of remaining alphas).
307 obs_sum -= obs(a);
308 alp_sum -= alpha(a);
309 vector<Type> obs_a(2), alphas_a(2);
310 obs_a(0) = obs(a); obs_a(1) = obs_sum;
311 alphas_a(0) = alpha(a); alphas_a(1) = alp_sum;
312 ll += k(a) * ddirmultinom(obs_a, alphas_a, 1);
313 }
314 } else {
315 ll = ddirmultinom(obs, alpha, 1);
316 }
317 return give_log ? ll : exp(ll);
318}
