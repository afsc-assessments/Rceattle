15{

26// Function to maintain > limit
27template<class Type>
28Type posfun(Type x, Type eps, Type &penalty) {
29 Type denom = 2;
30 denom -= x/eps;
31 Type ans = CppAD::CondExpGe(x, eps, x, eps/denom); // tune a new candidate x given eps
32 penalty += CppAD::CondExpGe(x, eps, Type(0), 0.01 * (x - eps) * (x - eps)); // penalize via x vs eps
33 return ans;
34}

37// Dirichlet multinomial (from WHAM)
38template<class Type>
39Type ddirmultinom(vector<Type> obs, vector<Type> alpha, int do_log)
40{
41 int dim = obs.size();
42 Type N = obs.sum();
43 Type phi=sum(alpha);
44 Type ll = lgamma(N + 1.0) + lgamma(phi) - lgamma(N + phi);
45 for(int a = 0; a < dim; a++) ll += -lgamma(obs(a) + 1.0) + lgamma(obs(a) + alpha(a)) - lgamma(alpha(a));
46 if(do_log == 1) return ll;
47 else return exp(ll);
48}

75/**
76 * @brief Concentrated (analytical) sd of a lognormal observation likelihood.
77 *
78 * Catch and index observations are fit as
79 * `log(obs) ~ N(log(pred) - b * sigma^2 / 2, sigma)`, where `b` is the
80 * bias-adjustment flag `bias_adjust_obs` (1 = predictions are mean-unbiased,
81 * Rceattle's default; 0 = median-unbiased). Minimising that density over sigma
82 * with `S` = mean squared log residual gives
83 *
84 * sigma^2 = 2 * S / (sqrt(1 + b^2 * S) + 1)
85 *
86 * which is the Ludwig and Walters (1994) estimator `sigma = sqrt(S)` when
87 * `b = 0`, and shrinks it as the bias term takes up part of the residual. The
88 * form above is the rationalised version of `2 (sqrt(1 + b^2 S) - 1) / b^2`;
89 * it needs no branch at `b = 0` and stays accurate for small `b`, where that
90 * expression cancels to zero.
91 *
92 * @param mean_sq_resid S, the mean of squared `log(obs) - log(pred)`.
93 * @param bias_adjust b, the bias-adjustment flag applied by the density.
94 * @return The sd that minimises the lognormal negative log-likelihood.
95 */
96template <class Type>
97Type concentrated_lognormal_sd(Type mean_sq_resid, Type bias_adjust)
98{
99 return sqrt( Type(2.0) * mean_sq_resid /
100 (sqrt(Type(1.0) + square(bias_adjust) * mean_sq_resid) + Type(1.0)) );
101}

104// Function to calculate the first difference
105template <class Type>
106vector<Type> first_difference(const vector<Type> &x)
107{
108 int length = x.size() - 1;
109
110 if(length <= 0){
111 std::cerr << "Error -- vector size too small in first_difference()" << std::endl;
112 return vector<Type>(0); // explicit empty vector (was an ambiguous `return(0)`)
113 }
114
115 vector<Type> tmp(length);
116
117 for(int i = 0; i < length; i++){
118 tmp(i) = x(i+1) - x(i);
119 }
120
121 return tmp;
122}

134/**
135 * @brief log(mean(exp(x))) over a non-empty vector, computed so it cannot overflow.
136 *
137 * The non-parametric selectivity forms rescale their log-scale coefficients
138 * against log(mean(exp(coefficients))) once per year, which is how a curve whose
139 * level is otherwise unidentified is held at mean selectivity 1. Summed directly,
140 * exp() overflows to infinity once a coefficient passes about 709 -- the largest
141 * argument it takes in double precision -- and the objective is NaN from there on.
142 * Subtracting the largest coefficient first holds every exponent at or below 0, so
143 * the sum is at most the number of bins whatever the coefficients are.
144 *
145 * The shift cancels out of both the result and its derivative.
146 *
147 * @param x non-empty vector of log-scale values.
148 * @return log of the mean of exp(x).
149 */
150template <class Type>
151Type log_mean_exp(const vector<Type> &x)
152{
153 Type mx = x(0);
154 for(int i = 1; i < x.size(); i++) mx = max2(mx, x(i));
155
156 Type total = 0;
157 for(int i = 0; i < x.size(); i++) total += exp(x(i) - mx);
158
159 return mx + log(total / Type(x.size()));
160}

163/**
164 * @brief Construct a sparse GMRF precision matrix separable along year, age, and cohort.
165 *
166 * Builds Q = (I - B^T) * Omega * (I - B), where B carries the partial
167 * correlations by year (rho_y), age (rho_a), and cohort (rho_c), and Omega is
168 * the diagonal (co)variance. `Var_Param` selects the conditional (0) or marginal
169 * (1) variance parameterization of the GMRF.
170 *
171 * @param n_years Number of years.
172 * @param n_ages Number of ages.
173 * @param ay_Index [total_n x 2] age/year index locating each node.
174 * @param rho_y Partial correlation across years.
175 * @param rho_a Partial correlation across ages.
176 * @param rho_c Partial correlation across cohorts.
177 * @param log_sigma2 Log variance parameter governing the GMRF.
178 * @param Var_Param 0 = conditional variance, 1 = marginal variance.
179 * @return Sparse precision matrix [total_n x total_n], total_n = n_years * n_ages.
180 */
181template<class Type>
182Eigen::SparseMatrix<Type> construct_Q(int n_years, // Integer of years
183 int n_ages, // Integer of ages
184 matrix<Type> ay_Index, // Index matrix to construct
185 Type rho_y, // Partial correlation by years
186 Type rho_a, // Partial correlation by ages
187 Type rho_c, // Partial correlation by cohort
188 Type log_sigma2, // Variance parameter governing GMRF
189 int Var_Param // Parameterization of Variance ==0 (Conditional), == 1(Marginal)
190) {
191
192 // Dimension to construct matrices
193 int total_n = n_years * n_ages;
194
195 // Construct matrices for precision matrix
196 Eigen::SparseMatrix<Type> B(total_n,total_n); // B matrix
197 Eigen::SparseMatrix<Type> I(total_n,total_n); // Identity matrix
198 I.setIdentity(); // Set I to identity matrix
199 Eigen::SparseMatrix<Type> Omega(total_n,total_n); // Omega matrix (variances)
200 Eigen::SparseMatrix<Type> Q_sparse(total_n, total_n); // Precision matrix
201
202 for(int n = 0; n < total_n; n++) {
203
204 // Define year and age objects
205 Type age = ay_Index(n,0);
206 Type year = ay_Index(n,1);
207
208 // Constructing B matrix to determine where the correlation pars should go
209 if(age > 1) {
210
211 // Get column index for years
212 for(int n1 = 0; n1 < total_n; n1++) {
213 if(ay_Index(n1, 0) == age - 1 && ay_Index(n1, 1) == year)
214 B.coeffRef(n, n1) = rho_y;
215 } // n1 loop
216
217 } // end age > 1
218
219 if(year > 1) {
220
221 // Get column index for years
222 for(int n1 = 0; n1 < total_n; n1++) {
223 if(ay_Index(n1, 0) == age && ay_Index(n1, 1) == year - 1)
224 B.coeffRef(n, n1) = rho_a;
225 } // n1 loop
226
227 } // if year > 1
228
229 if(year > 1 && age > 1) {
230
231 // Get column index for years
232 for(int n1 = 0; n1 < total_n; n1++) {
233 if(ay_Index(n1, 0) == age - 1 && ay_Index(n1, 1) == year - 1)
234 B.coeffRef(n,n1) = rho_c; // correlation by cohort
235 } // n1 loop
236
237 } // if both year and age > 1
238
239 } // end n loop
240
241 // Fill in Omega matrix here (variances)
242 if(Var_Param == 0) { // Conditional variance
243
244 for(int i = 0; i < total_n; i++) {
245 for(int j = 0; j < total_n; j++) {
246 if(i == j) Omega.coeffRef(i,j) = 1/exp(log_sigma2);
247 else Omega.coeffRef(i,j) = Type(0.0);
248 } // j loop
249 } // i loop
250
251 } // end if conditional variance
252
253 if(Var_Param == 1) { // Marginal Variance
254
255 // Construct container objects
256 matrix<Type> L(total_n, total_n); // L Matrix
257 matrix<Type> tmp_I_B = I-B; // Temporary Matrix to store I-B
258 L = tmp_I_B.inverse(); // Invert to get L
259 vector<Type> d(total_n); // Store variance calculations
260
261 for(int n = 0; n < total_n; n++) {
262 if(n == 0) {
263 d(n) = exp(log_sigma2); // marginal variance parameter
264 } else{
265
266 Type cumvar = 0; // Cumulative Variance Container
267
268 for(int n1 = 0; n1 < n; n1++) {
269 cumvar += L(n, n1) * d(n1) * L(n, n1);
270 } // n1 loop
271
272 // Calculate diagonal values for omega
273 d(n) = (exp(log_sigma2) - cumvar) / pow(L(n, n), 2);
274
275 } // else loop
276 } // n loop
277
278 // Now fill in our diagonals for Omega
279 for(int i = 0; i < total_n; i++) {
280 for(int j = 0; j < total_n; j++) {
281 if(i == j) Omega.coeffRef(i,j) = 1/d(i);
282 else Omega.coeffRef(i,j) = Type(0.0);
283 } // j loop
284 } // i loop
285
286 } // end if marginal variance
287
288 // Now, do calculations to construct (Q = (I - t(B)) %*% Omega %*% (I-B))
289 Eigen::SparseMatrix<Type> B_transpose = B.transpose(); // transpose B matrix
290
291 // Calculate Precision Matrix
292 Q_sparse = (I - B_transpose) * Omega * (I-B);
293
294 return(Q_sparse);
295
296} // end construct_Q function
