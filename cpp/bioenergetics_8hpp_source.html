4/**
5 * @brief Calculates the temperature-dependent function of consumption (fT) for bioenergetics.
6 *
7 * Updates the fT matrix in place. Supports multiple bioenergetics models:
8 * 1 = Exponential function (Stewart et al. 1983)
9 * 2 = Warm-water species temperature dependence (Kitchell et al. 1977)
10 * 3 = Cool/cold-water species temperature dependence (Thornton and Lessem 1979)
11 * 4 = Constant (fT = 1.0)
12 *
13 * @param fT Matrix of temperature functions to be updated [nspp, nyrs].
14 * @param nspp Number of species.
15 * @param nyrs Number of years.
16 * @param Ceq Vector specifying which consumption equation to use for each species.
17 * @param Cindex Vector specifying which environmental index drives each species.
18 * @param Qc Bioenergetics parameter (QC value / intercept).
19 * @param Tcm Bioenergetics parameter (Thermal max).
20 * @param Tco Bioenergetics parameter (Thermal optimum).
21 * @param Tcl Bioenergetics parameter (Thermal limit for Thornton/Lessem).
22 * @param CK1 Bioenergetics parameter (Ascending limit for Thornton/Lessem).
23 * @param CK4 Bioenergetics parameter (Descending limit for Thornton/Lessem).
24 * @param env_index Matrix of environmental predictors (e.g., bottom temperature) [nyrs, n_indices].
25 */
26template <class Type>
27void calculate_temperature_function(matrix<Type> &fT,
28 int nspp,
29 int nyrs,
30 const vector<int> &Ceq,
31 const vector<int> &Cindex,
32 const vector<Type> &Qc,
33 const vector<Type> &Tcm,
34 const vector<Type> &Tco,
35 const vector<Type> &Tcl,
36 const vector<Type> &CK1,
37 const vector<Type> &CK4,
38 const matrix<Type> &env_index) {
39
40 // Pre-allocate temporary variables used in Kitchell and Thornton/Lessem equations
41 // Hoisted out of the loops to optimize memory allocation on the AD tape
42 Type Yc = 0.0;
43 Type Zc = 0.0;
44 Type Vc = 0.0;
45 Type Xc = 0.0;
46 Type G2 = 0.0;
47 Type L2 = 0.0;
48 Type G1 = 0.0;
49 Type L1 = 0.0;
50 Type Ka = 0.0;
51 Type Kb = 0.0;
52
53 for(int sp = 0; sp < nspp; sp++) {
54
55 // Equations 1-3 read one column of env_index; equation 4 reads none. Cindex
56 // is 0-based here (ceattle.cpp decrements the 1-based input). With no column
57 // to read, consumption takes equation 4's constant fT = 1 -- the model is
58 // built without bounds checking, so the read would return adjacent memory.
59 bool has_env = (Cindex(sp) >= 0) && (Cindex(sp) < env_index.cols());
60 int eq = has_env ? Ceq(sp) : 4;
61
62 for(int yr = 0; yr < nyrs; yr++) {
63
64 switch(eq) {
65 case 1: // Exponential function from Stewart et al. 1983
66 fT(sp, yr) = exp(Qc(sp) * env_index(yr, Cindex(sp)));
67 break;
68
69 case 2: // Temperature dependence for warm-water species from Kitchell et al. 1977
70 Yc = log(Qc(sp)) * (Tcm(sp) - Tco(sp) + 2.0);
71 Zc = log(Qc(sp)) * (Tcm(sp) - Tco(sp));
72 Vc = (Tcm(sp) - env_index(yr, Cindex(sp))) / (Tcm(sp) - Tco(sp));
73 Xc = pow(Zc, 2) * pow((1.0 + pow((1.0 + 40.0 / Yc), 0.5)), 2) / 400.0;
74 fT(sp, yr) = pow(Vc, Xc) * exp(Xc * (1.0 - Vc));
75 break;
76
77 case 3: // Temperature dependence for cool and cold-water species from Thornton and Lessem 1979
78 G2 = (1.0 / (Tcl(sp) - Tcm(sp))) * log((0.98 * (1.0 - CK4(sp))) / (CK4(sp) * 0.02));
79 L2 = exp(G2 * (Tcl(sp) - env_index(yr, Cindex(sp))));
80 Kb = (CK4(sp) * L2) / (1.0 + CK4(sp) * (L2 - 1.0));
81
82 G1 = (1.0 / (Tco(sp) - Qc(sp))) * log((0.98 * (1.0 - CK1(sp))) / (CK1(sp) * 0.02));
83 L1 = exp(G1 * (env_index(yr, Cindex(sp)) - Qc(sp)));
84 Ka = (CK1(sp) * L1) / (1.0 + CK1(sp) * (L1 - 1.0));
85
86 fT(sp, yr) = Ka * Kb;
87 break;
88
89 case 4: // Constant
90 fT(sp, yr) = 1.0;
91 break;
92
93 default:
94 // Ceq outside 1-4 (data_check() refuses it). fT = 1 rather than the zero
95 // left by initialization, which would silently zero consumption and,
96 // under multispecies, predation mortality with it.
97 fT(sp, yr) = 1.0;
98 break;
99 }
100 }
101 }

108 * This function calculates the amount of food consumed by a predator of a specific

110 * arrays in place. It supports either full bioenergetics equations (Ceq < 4) or
111 * direct empirical input of annual ration (Ceq == 4).
112 *
113 * @param consumption_at_age Array updated with individual consumption (kg/yr) [nspp, max_sex, max_age, nyrs].
114 * @param nspp Total number of species in the model.
115 * @param nyrs Total number of years (hindcast + forecast).
116 * @param nyrs_hind Number of years in the hindcast period.
117 * @param nsex Vector containing the number of sexes modeled for each species.
118 * @param nages Vector containing the number of age classes for each species.
119 * @param Ceq Vector specifying the consumption equation type for each species.
120 * @param CA Vector of intercepts for the allometric mass function (Cmax = CA * W^(1+CB)).
121 * @param CB Vector of slopes for the allometric mass function.
122 * @param fday Vector of the number of foraging days per year for each species.
123 * @param Pvalue Vector of the proportion of maximum consumption (scales Cmax).
124 * @param fT Matrix of the temperature-dependence function [nspp, nyrs].
125 * @param ration_data Array of empirical relative-foraging rates or input ration limits.
