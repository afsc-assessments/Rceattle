4/**
5 * @brief SD of length-at-age (cm) for one age, year and sex.
6 *
7 * The variability term v is \f$e^{sd_0}\f$ at or below `age_L1`, pinned to
8 * \f$e^{sd_1}\f$ for the plus group under `sd_style == 1` (WHAM; also SS3 when
9 * Growth_Age_for_L2 = 999), and otherwise interpolated linearly in length
10 * between \f$(l_1, e^{sd_0})\f$ and \f$(L_\infty, e^{sd_1})\f$. `sd_form` says what v is:
11 * 1 = an SD in cm (SS3 CV_Growth_Pattern 2), 2 = a CV, so SD = v * L
12 * (SS3 CV_Growth_Pattern 0).
13 */
14template<class Type>
15Type length_sd_at_age(Type current_age, Type age_L1, bool plus_group,
16 int sd_style, int sd_form, Type l1, Type linf, Type len,
17 Type log_sd0, Type log_sd1) {
18 Type sd0 = exp(log_sd0);
19 Type sd1 = exp(log_sd1);
20 Type v;
21 if(current_age <= age_L1) {
22 v = sd0;
23 } else if(plus_group && sd_style == 1) {
24 v = sd1;
25 } else {
26 v = sd0 + (sd1 - sd0) / (linf - l1) * (len - l1);
27 }
28 if(sd_form == 2) v = v * len;
29 return v;
30}

33/**
34 * @brief Age-length key, weight-at-age and mature weight-at-age for one cell.
35 *
36 * Length is normal with mean `mu` and SD `sd` (cm). Probabilities are taken on
37 * the population length bins `lengths_pop` (lower edges): the first bin is a
38 * minus group below the second edge and the last a plus group above the last
39 * edge (SS3's convention). They stay on that grid; the consumers that need data
40 * bins sum each bin's run of population bins themselves.
41 *
42 * Weight-at-age is \f$\sum_l P(l|a) \alpha L_{mid}^\beta\f$ at population-bin
43 * midpoints (kg); the last bin's midpoint sits half a bin width above its lower
44 * edge. When the species has maturity-at-length, mature weight-at-age is the
45 * same sum with the logistic maturity \f$1/(1+e^{-s(L_{mid} - L_{50})})\f$ inside:
46 * SS3's fecundity-at-age when fecundity equals body weight.
47 */
48template<class Type>
49void fill_age_length_key(int wtind, int sp, int sex, int age, int yr,
50 Type mu, Type sd,
51 const vector<int>& nlengths_pop,
52 matrix<Type>& lengths_pop,
53 matrix<Type>& weight_length_pars,
54 const vector<int>& mat_len_use,
55 matrix<Type>& mat_len_pars,
56 array<Type>& growth_matrix,
57 array<Type>& weight_hat,
58 array<Type>& mat_weight_hat) {
59 int np = nlengths_pop(sp);
60 // Kept on the POPULATION bins. Its readers multiply it by
61 // selectivity-at-length, and SS3 forms that product at population resolution
62 // before binning, so aggregating here would apply one bin-average selectivity
63 // across a data bin. ceattle.cpp sums each data bin's run of population bins
64 // (section 2.3c) where it needs data bins.
65 for(int lp = 0; lp < np; lp++) growth_matrix(wtind, sex, age, lp, yr) = Type(0.0);
66
67 Type expected_weight = 0.0;
68 Type expected_mat_weight = 0.0;
69 for(int lp = 0; lp < np; lp++) {
70 Type prob;
71 if(np < 2) {
72 // One bin spans the whole length distribution, so it holds all of it.
73 // Tested before the lp == 0 branch, which would read the second edge.
74 prob = Type(1.0);
75 } else if(lp == 0) {
76 prob = pnorm((lengths_pop(sp, 1) - mu) / sd);
77 } else if(lp == (np - 1)) {
78 prob = 1.0 - pnorm((lengths_pop(sp, np - 1) - mu) / sd);
79 } else {
80 prob = pnorm((lengths_pop(sp, lp + 1) - mu) / sd) - pnorm((lengths_pop(sp, lp) - mu) / sd);
81 }
82 growth_matrix(wtind, sex, age, lp, yr) = prob;
83
84 // Midpoint (cm) of the bin: lower edge plus half its own width, the last bin
85 // taking the previous bin's width. A single-bin grid has no neighbour to take
86 // a width from, so it keeps the edge itself -- as selectivity.hpp does.
87 Type lenmid;
88 if(lp < np - 1) {
89 lenmid = (lengths_pop(sp, lp) + lengths_pop(sp, lp + 1)) / Type(2.0);
90 } else if(np < 2) {
91 lenmid = lengths_pop(sp, np - 1);
92 } else {
93 lenmid = lengths_pop(sp, np - 1) + (lengths_pop(sp, np - 1) - lengths_pop(sp, np - 2)) / Type(2.0);
94 }
95 Type wt_len = weight_length_pars(sp, 0) * pow(lenmid, weight_length_pars(sp, 1));
96 expected_weight += prob * wt_len;
97 if(mat_len_use(sp) == 1) {
98 expected_mat_weight += prob * wt_len / (Type(1.0) + exp(-mat_len_pars(sp, 1) * (lenmid - mat_len_pars(sp, 0))));
99 }
100 }
101 weight_hat(wtind, sex, age, yr) = expected_weight;
102 if(mat_len_use(sp) == 1) mat_weight_hat(wtind, sex, age, yr) = expected_mat_weight;
103}

105/**
106 * @brief Integrated Growth, Size-Transition, and Weight-at-Age Module for Month = 0.
107 *
108 * Computes Jan-1 mean length-at-age, the age->length probability matrix, and
109 * integrated weight-at-age for a single species.
110 *
111 * @section math_models Mathematical Models:
112 * 1. Mean Length-at-Age (\f$L_a\f$) — anchored at \f$L(a_{L1}) = l_1\f$:
113 * - Von Bertalanffy (Model 1): \f$L_a = L_{\infty} + (l_1 - L_{\infty}) \cdot e^{-K(a - a_{L1})}\f$
114 * - Richards (Model 2): \f$L_a = [L_{\infty}^m + (l_1^m - L_{\infty}^m) \cdot e^{-K(a - a_{L1})}]^{1/m}\f$
115 * - For year > 0, a cohort recursion advances \f$L_{a, y}\f$ from \f$L_{a-1, y-1}\f$
116 * using lag-year parameters. At the cohort boundary (current_age ==
117 * age_L1_ceil) the closed-form anchor at \f$l_1\f$ is used; `age_L1_safe`
118 * keeps this anchor at \f$l_1\f$ for both minage > 0 and minage = 0.
119 * - Linear ramp from Lmin_sp at age 0 to \f$l_1\f$ at age_L1 applies when
120 * current_age <= age_L1 (only reachable for minage > 0).
121 *
122 * 2. Weight-at-Age (\f$W_a\f$):
123 * Integrated across length to account for Jensen's Inequality:
124 * \f$W_a = \sum_{ln} P(ln | a) \cdot \alpha \cdot L_{mid}^{\beta}\f$
125 * Bin midpoints are computed per-bin to support non-uniform length bins;
126 * the length plus-group is extended by half the final interior bin width.
127 *
128 * @section logic Biological Logic:
129 * - **Temporal Resolution**: Jan-1 (month = 0).
130 * - **Plus-Age Group**: The oldest age's mean length is set by
131 * `growth_plus_length`: an M1-weighted mean toward \f$L_{\infty}\f$ (1, the
132 * default), no adjustment (2), or SS3's Linf_decay forms (3 = -999, 4 = a
133 * decay rate). See section 2 of the body.
134 * - **SD-at-Age**: length_sd_at_age(); `growth_sd_form` makes the two
135 * endpoints SDs (cm) or CVs.
136 * - **Size Transition**: fill_age_length_key(), on the population length
137 * bins; consumers bin the product with selectivity, not the key itself.
138 *
139 * @param wtind Weight index slot to write into.
140 * @param sp Species index.
141 * @param nyrs Number of years in the simulation/model.
142 * @param nsex/nages/nlengths Vectors of species-specific counts per dimension.
143 * @param minage Vector of minimum modeled ages (0 supported).
144 * @param growth_model Per-species growth model selector (1 = VBGF, 2 = Richards).
145 * @param lengths Matrix of length bin boundaries (nspp x length_bins). Bin
146 * widths may be non-uniform.
147 * @param growth_parameters 4D array of time-varying growth parameters (K, L1, Linf, m).
148 * @param growth_log_sd 3D array of log-scale SDs (index 0 = SD at L1, index 1 = SD at Linf).
149 * @param weight_length_pars Matrix of length-weight parameters (\f$\alpha\f$, \f$\beta\f$).
150 *
151 * @note Outputs are passed by reference.
152 * @param length_hat [Output] 4D array filled with mean length-at-age.
153 * @param growth_matrix [Output] 5D array filled with the growth transition matrix.
154 * @param weight_hat [Output] 4D array filled with integrated weight-at-age.
155 */

169 const vector<int>& growth_sd_form, // 1 = SD endpoints in cm, 2 = CV endpoints (SD = CV * L)
170 const vector<int>& growth_plus_length,// Plus-group mean length: 1 = M1-weighted, 2 = none, 3 = SS3.24, 4 = decay

172 matrix<Type>& lengths,
173 const vector<int>& nlengths_pop,
174 matrix<Type>& lengths_pop, // Population length bins, lower edges (cm)
175 array<Type>& growth_parameters,
176 array<Type>& growth_log_sd,
177 matrix<Type>& weight_length_pars,
178 const vector<int>& mat_len_use, // 1 = maturity-at-length for this species
179 matrix<Type>& mat_len_pars, // [sp, 0] = L50 (cm), [sp, 1] = logistic slope (per cm)
180 array<Type>& log_M1, // Base natural mortality at age [nspp, nsex, nages], log scale
181 array<Type> &length_hat, // Modified by reference
182 array<Type> &growth_matrix, // Modified by reference
183 array<Type> &weight_hat, // Modified by reference
184 array<Type> &mat_weight_hat // Modified by reference (maturity-at-length species only)

188

190 // Calculate mean-length, SD, and growth matrix, for all years:
191 // lengths is vector with lengths mm (2, 4, 6, 8, etc)
192 Type b_len, last_linear, current_age;
193
194 // The linear ramp below age_L1 starts from the lowest population length
195 // edge at age 0 (SS3's len_bins(1)); with no population grid that is the
196 // lowest data bin.
197 Type Lmin_sp = lengths_pop(sp, 0);
198 // age_L1 is the VB anchor age (= age at which `l1` is the length). Read
199 // from data_list$growth_age_L1[sp] (= SS3's Growth_Age_for_L1 ctl input).
200 // R-side fit_mod() resolves the default to max(0.5, minage[sp]) so models
201 // with minage = 0 get an SS3-style half-year anchor and minage >= 1 stays
202 // backwards-compatible.
203 Type age_L1 = growth_age_L1(sp);
204 // age_L1_ceil is compared to current_age (the slot age in years) to detect
205 // the youngest VB-relevant slot. That slot needs the closed-form boundary
206 // formula (not the cohort recursion which would index age-1 = -1 at slot 0
207 // when minage = 0, or use a stale value at slot 1 when the anchor is at
208 // age 0.5). For minage = 0, the youngest VB slot is C++ age = 1
209 // (current_age = 1, anchor at 0.5). For minage >= 1, slot 0 itself
210 // (current_age = minage). Either way: age_L1_ceil = max(1, minage).
211 Type age_L1_ceil = (Type(minage(sp)) >= Type(1)) ? Type(minage(sp)) : Type(1);
212
213 // When minage = 0, the linear ramp from Lmin_sp at age 0 to l1 at age_L1
214 // becomes degenerate (divide by zero in b_len) AND the ramp branch should
215 // never execute (current_age >= 1 always > age_L1 = 0). Replace the ramp
216 // anchor with a small positive value so b_len is finite; the value is
217 // unused because the if(current_age <= age_L1) branch is unreachable.
218 Type age_L1_safe = age_L1;
219 if (age_L1 <= Type(0)) age_L1_safe = Type(1); // safe denominator only
220
221 for(int sex = 0; sex < nsex(sp); sex++) {
222 for(int yr = 0; yr < nyrs; yr++) {
223 for(int age = 0; age < nages(sp); age++) {
224
225 // Parameters for parametric growth
226 Type kappa = growth_parameters(sp, sex, yr, 0);
227 Type l1 = growth_parameters(sp, sex, yr, 1);
228 Type linf = growth_parameters(sp, sex, yr, 2);
229 Type m = growth_parameters(sp, sex, yr, 3);
230 // current_age is the slot's actual age (start of year). Slot index
231 // age (0-based) holds the (age + minage) cohort. The historical
232 // `age + 1.0` shifted everything by one year at any minage != 1
233 // (gave end-of-year length instead of start). Note this is the SLOT
234 // age, NOT the VB anchor (which is age_L1, defined above).
235 current_age = Type(age) + Type(minage(sp));
236
237 // 1. Calculate Mean Length at Age ---
238 switch(growth_model(sp)) {
239 case 1: // Von Bertalanffy
240
241 // Slope from Lmin to L1
242 b_len = (l1 - Lmin_sp) / age_L1_safe; // age_L1_safe == age_L1 when minage > 0; == 1 when minage = 0 (ramp branch unreachable)
243
244 // Age < minage
245 if((current_age) <= age_L1){
246 length_hat(wtind, sex, age, yr) = Lmin_sp + b_len * (current_age);
247 } else {
248 if(yr == 0) {
249 length_hat(wtind, sex, age, yr) = linf + (l1 - linf) * exp(-kappa * (current_age - age_L1));
250 } else { // Yr > 0
251 if(current_age == age_L1_ceil) {
252 // age_L1_safe ensures last_linear == l1 in both minage > 0
253 // (where age_L1_safe == age_L1) and minage = 0 (where
254 // age_L1 = 0 would otherwise collapse this to Lmin_sp).
255 last_linear = Lmin_sp + b_len * age_L1_safe;
256 length_hat(wtind, sex, age, yr) = last_linear + (last_linear - linf) * (exp(-kappa * (current_age - age_L1)) - 1.0);
257 } else {
258 // Lag 1-year parameters
259 Type lagkappa = growth_parameters(sp, sex, yr - 1, 0);
260 Type laglinf = growth_parameters(sp, sex, yr - 1, 2);
261 length_hat(wtind, sex, age, yr) = length_hat(wtind, sex, age - 1, yr - 1) + (length_hat(wtind, sex, age - 1, yr - 1) - laglinf) * (exp(-lagkappa) - 1.0);
262 }
263 }
264 }
265 break;
266
267 case 2: // Richards
268
269 // Slope from Lmin to L1
270 b_len = (l1 - Lmin_sp) / age_L1_safe; // age_L1_safe == age_L1 when minage > 0; == 1 when minage = 0 (ramp branch unreachable)
271
272 if((current_age) <= age_L1) {
273 length_hat(wtind, sex, age, yr) = Lmin_sp + b_len * (current_age);
274 } else {
275 if(yr == 0) {
276 length_hat(wtind, sex, age, yr) = pow(pow(linf, m) + (pow(l1, m) - pow(linf, m)) * exp(-kappa * (current_age - age_L1)), 1 / m);
277 } else { // Yr > 0
278 if(current_age == age_L1_ceil) {
279 // age_L1_safe ensures last_linear == l1 in both minage > 0
280 // and minage = 0; see VBGF branch above for full reasoning.
281 last_linear = Lmin_sp + b_len * age_L1_safe;
282 length_hat(wtind, sex, age, yr) = pow(pow(last_linear, m) + (pow(last_linear, m) - pow(linf, m)) * (exp(-kappa * (current_age - age_L1)) - 1.0), 1 / m);
283 } else {
284 // Lag 1-year parameters
285 Type lagkappa = growth_parameters(sp, sex, yr - 1, 0);
286 Type laglinf = growth_parameters(sp, sex, yr - 1, 2);
287 Type lagm = growth_parameters(sp, sex, yr - 1, 3);
288 length_hat(wtind, sex, age, yr) = pow(pow(length_hat(wtind, sex, age - 1, yr - 1), lagm) + (pow(length_hat(wtind, sex, age - 1, yr - 1), lagm) - pow(laglinf, lagm)) * (exp(-lagkappa) - 1.0), 1 / lagm);
289 }
290 }
291 }
292 break;
293
294 case 3: // Non-parametric (Free parameters)
295 error("Non-parametric growth not yet implemented");
296 // length_hat(wtind, sex, age, yr) = exp(length_par(sp, sex, age) + length_par_re(sp, sex, age, yr));
297 break;
298
299 default:
300 error("Invalid 'growth_model");
301 } // Growth_model switch
302
303
304 // 2. Plus-Group Mean Length (Oldest Age Only) ---
305 // The plus group holds fish older than the oldest age, so its mean length
306 // (cm) sits between L(oldest age) and L-infinity. growth_plus_length picks
307 // the form: 1 = M1-weighted, 2/3/4 = SS3's Linf_decay -998/-999/d. See
308 // ?build_growth.
309 if(growth_model(sp) < 3 && age == (nages(sp) - 1)) {
310 Type current_size = length_hat(wtind, sex, age, yr);
311 Type diff = linf - current_size;
312 int oldest_age = minage(sp) + nages(sp) - 1;
313 if(growth_plus_length(sp) == 1) {
314 Type temp_n = 0, temp_sum = 0, weight_a = 1.0;
315 Type surv = exp(-exp(log_M1(sp, sex, nages(sp) - 1)));
316 for(int a = 0; a <= nages(sp); a++) {
317 temp_sum += weight_a * (current_size + (Type(a) / Type(nages(sp))) * diff);
318 temp_n += weight_a;
319 weight_a *= surv;
320 }
321 length_hat(wtind, sex, age, yr) = temp_sum / temp_n;
322 } else if(growth_plus_length(sp) == 3) {
323 Type temp_n = 0, temp_sum = 0;
324 // Survival weights at a fixed 0.2/yr, the nominal mortality SS3 uses
325 // for Linf_decay = -999 -- not this species' estimated M1, which is
326 // what form 1 above weights by.
327 for(int a = 0; a <= oldest_age; a++) {
328 Type weight_a = exp(Type(-0.2) * Type(a));
329 temp_sum += weight_a * (current_size + (Type(a) / Type(oldest_age)) * diff);
330 temp_n += weight_a;
331 }
332 length_hat(wtind, sex, age, yr) = temp_sum / temp_n;
333 } else if(growth_plus_length(sp) == 4) {
334 Type size = current_size, temp_sum = current_size, temp_n = 1.0, weight_a = 1.0;
335 // Averaged over twice the oldest age, far enough past it that the
336 // decay weights leave no meaningful length behind (SS3's own span).
337 for(int a = 1; a <= 2 * oldest_age; a++) {
338 weight_a *= exp(-plus_group_decay(sp));
339 size += (linf - size) * (Type(1.0) - exp(-kappa));
340 temp_sum += weight_a * size;
341 temp_n += weight_a;
342 }
343 length_hat(wtind, sex, age, yr) = temp_sum / temp_n;
344 }
345 }
346
347 // 3. SD of length-at-age, then age-length key and weight-at-age ---
348 if(growth_model(sp) < 3) {
349 Type len = length_hat(wtind, sex, age, yr);
350 Type sd = length_sd_at_age(current_age, age_L1, age == (nages(sp) - 1),
351 growth_sd_style(sp), growth_sd_form(sp), l1, linf, len,
352 growth_log_sd(sp, sex, 0), growth_log_sd(sp, sex, 1));
353 fill_age_length_key(wtind, sp, sex, age, yr, len, sd, nlengths_pop,
354 lengths_pop, weight_length_pars,
355 mat_len_use, mat_len_pars, growth_matrix, weight_hat, mat_weight_hat);
356 }
357 } // age
358 } // yr
359 } // sex

373 *

379 * - **Temporal Resolution**: Incorporates `fracyr` to allow within-year (seasonal)

387 * consumers bin the product with selectivity, not the key itself.
388 *
389 * @param wtind Weight index for population/fleet.
390 * @param id_pop Index for population (Jan-1) weight-at-age; the within-year
391 * growth advance is anchored on `length_hat(id_pop, ...)`.
392 * @param sp Species index.
393 * @param fracyr Fraction of the year (0.0 - 1.0) for within-year growth.
394 * @param nyrs Number of years in the simulation/model.
395 * @param nsex/nages/nlengths Vectors of species-specific counts per dimension.
396 * @param lengths Matrix of length bin boundaries (nspp x length_bins). Bin
397 * widths may be non-uniform.
398 * @param growth_parameters 4D array of time-varying growth parameters (K, L1, Linf, m).

402 * @note Outputs are passed by reference.
403 * @param length_hat [Output] 4D array filled with mean length-at-age.
404 * @param growth_matrix [Output] 5D array filled with the growth transition matrix.
405 * @param weight_hat [Output] 4D array filled with integrated weight-at-age.

450 // from data_list$growth_age_L1[sp] (= SS3's Growth_Age_for_L1 ctl input).
451 // R-side fit_mod() resolves the default to max(0.5, minage[sp]) so models
452 // with minage = 0 get an SS3-style half-year anchor and minage >= 1 stays
453 // backwards-compatible.
454 Type age_L1 = growth_age_L1(sp);
455
456 // Safe denominator for the linear ramp slope when minage = 0. See month=0
457 // estimate_growth() for full context.
458 Type age_L1_safe = age_L1;
459 if (age_L1 <= Type(0)) age_L1_safe = Type(1);
460
461 for(int sex = 0; sex < nsex(sp); sex++) {
462 for(int yr = 0; yr < nyrs; yr++) {
463 for(int age = 0; age < nages(sp); age++) {
464
465 // Parameters for parametric growth
466 Type kappa = growth_parameters(sp, sex, yr, 0);
467 Type l1 = growth_parameters(sp, sex, yr, 1);
468 Type linf = growth_parameters(sp, sex, yr, 2);
469 Type m = growth_parameters(sp, sex, yr, 3);
470 // See month=0 overload: slot k = age (k - 1 + minage); current_age is
471 // slot age (NOT the VB anchor) offset by fracyr. Was
472 // `age + 1.0 + fracyr`, which is off by one when minage != 1.
473 current_age = Type(age) + Type(minage(sp)) + fracyr;
474
475 // 1. Calculate Mean Length at Age ---
476 switch(growth_model(sp)) {
477 case 1: // Von Bertalanffy
478
479 // Slope from Lmin to L1
480 b_len = (l1 - Lmin_sp) / age_L1_safe; // age_L1_safe == age_L1 when minage > 0; == 1 when minage = 0 (ramp branch unreachable)
481
482 // Age < minage
483 // The plus group is advanced through within-year growth like every
484 // other age (SS3 convention). The recruitment-weighted-mean
485 // correction for fish promoted into the plus group is reapplied at
486 // every year boundary by estimate_growth(),
487 // so id_pop already carries the corrected Jan-1 length.
488 if((current_age) <= age_L1){
489 length_hat(wtind, sex, age, yr) = Lmin_sp + b_len * (current_age);
490 // WHAM-style parameterization: for ages between the anchor and the
491 // plus group, blend the linear ramp with the growth curve; the plus
492 // group is pinned at its Jan-1 (id_pop) length rather than advanced by
493 // within-year growth.
494 // TODO(review): this branch (and its Richards mirror below) tests
495 // `age + 1.0 < age_L1`, mixing the slot index with an age, while the
496 // sibling branch above uses `current_age` (= age + minage + fracyr).
497 // For minage != 1 the two disagree, so the interpolated length can be
498 // off by one age. Unreachable for minage 0/1; verify for minage >= 2.
499 }else if(age + 1.0 < age_L1){ // Linear + growth curve mixed
500 last_linear = Lmin_sp + b_len * age_L1;
501 length_hat(wtind, sex, age, yr) = last_linear + (last_linear - linf) * (exp(-kappa * (current_age - age_L1)) - 1.0);
502 }else if(age + 1.0 == nages(sp) && growth_plus_length(sp) == 1) { // Plus group held at its Jan-1 length; under the SS3 forms it grows like every other age
503 length_hat(wtind, sex, age, yr) = length_hat(id_pop, sex, age, yr);
504 }else { // Growth curve
505 length_hat(wtind, sex, age, yr) = length_hat(id_pop, sex, age, yr) + (length_hat(id_pop, sex, age, yr) - linf) * (exp(-kappa * fracyr) - 1.0); // Add fracyr growth
506 }
507 break;
508
509 case 2: // Richards
510
511 // Slope from Lmin to L1
512 b_len = (l1 - Lmin_sp) / age_L1_safe; // age_L1_safe == age_L1 when minage > 0; == 1 when minage = 0 (ramp branch unreachable)
513
514 // Mirror the VBGF branch above with the Richards curve so the plus
515 // group is treated identically regardless of growth family: a linear
516 // ramp below the anchor, a mixed linear+growth branch, the plus group
517 // pinned at its Jan-1 (id_pop) length, and within-year Richards growth
518 // for the remaining ages. (`last_linear` is declared at function scope.)
519 if((current_age) <= age_L1){
520 length_hat(wtind, sex, age, yr) = Lmin_sp + b_len * (current_age);
521 }else if(age + 1.0 < age_L1){ // Linear + growth curve mixed
522 last_linear = Lmin_sp + b_len * age_L1;
523 length_hat(wtind, sex, age, yr) = pow(pow(last_linear, m) + (pow(last_linear, m) - pow(linf, m)) * (exp(-kappa * (current_age - age_L1)) - 1.0), 1 / m);
524 }else if(age + 1.0 == nages(sp) && growth_plus_length(sp) == 1) { // Plus group held at its Jan-1 length (see VBGF branch)
525 length_hat(wtind, sex, age, yr) = length_hat(id_pop, sex, age, yr);
526 } else {
527 length_hat(wtind, sex, age, yr) = pow(pow(length_hat(id_pop, sex, age, yr), m) + (pow(length_hat(id_pop, sex, age, yr), m) - pow(linf, m)) * (exp(-kappa * fracyr) - 1.0), 1 / m); // Add fracyr growth
528 }
529 break;
530
531 case 3: // Non-parametric (Free parameters)
532 error("Non-parametric growth not yet implemented");
533 // length_hat(wtind, sex, age, yr) = exp(length_par(sp, sex, age) + length_par_re(sp, sex, age, yr));
534 break;
535
536 default:
537 error("Invalid 'growth_model");
538 } // Growth_model switch
539
540 // 2. SD of length-at-age, then age-length key and weight-at-age ---
541 if(growth_model(sp) < 3) {
542 Type len = length_hat(wtind, sex, age, yr);
543 Type sd = length_sd_at_age(current_age, age_L1, age == (nages(sp) - 1),
544 growth_sd_style(sp), growth_sd_form(sp), l1, linf, len,
545 growth_log_sd(sp, sex, 0), growth_log_sd(sp, sex, 1));
546 fill_age_length_key(wtind, sp, sex, age, yr, len, sd, nlengths_pop,
547 lengths_pop, weight_length_pars,
548 mat_len_use, mat_len_pars, growth_matrix, weight_hat, mat_weight_hat);
549 }
550 } // age
551 } // yr
552 } // sex
553}
