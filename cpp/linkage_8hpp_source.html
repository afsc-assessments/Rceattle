1// =====================================================================
2// linkage.hpp
3//
4// Turns the linkage table (a covariate or time effect on a process parameter)
5// into year-by-year offsets that recruitment, M, growth, catchability and
6// selectivity add to their parameters. Each offset is beta * X(year, column).
7//
8// Example: build_M1(linkages = list(M1 = linkage_spec(~ temp))) makes two rows per
9// species, an (Intercept) held at 0 and a `temp` slope beta. rceattle_apply_M_linkages()
10// fills M_linkage_offset(sp, sex, age, yr) = beta * temp_yr, and ceattle.cpp section 6.1
11// reads M1_at_age = exp(log M1 + M_linkage_offset). The equations for every link and
12// the row / slot / group terms are at ceattle.cpp section 2.3b.
13//
14// Process and parameter codes are in lockstep with R/0-linkage_encode.R
15// (LINKAGE_PROCESS_CODES / LINKAGE_PARAM_CODES); change one, change both.
16// A stratum id of 0 (species, sex, age bin or fleet) means "every level".
17// =====================================================================
18#ifndef RCEATTLE_LINKAGE_HPP
19#define RCEATTLE_LINKAGE_HPP

73/**
74 * @brief Expand a stratum id into the loop range [lo, hi).
75 *
76 * Clamped to `n_levels`, so an id past the last level gives an empty range.
77 *
78 * @param id 1-based stratum id, or 0 for "all levels".
79 * @param n_levels Number of levels the caller loops over.
80 * @param lo,hi [out] First level, and one past the last level.
81 */
82inline void rceattle_stratum_range(int id, int n_levels, int& lo, int& hi) {
83 lo = (id == 0) ? 0 : (id - 1);
84 hi = (id == 0) ? n_levels : id;
85 if (hi > n_levels) hi = n_levels;
86}

89/**
90 * @brief The linkage table: one entry per row, and the shared design matrix.
91 *
92 * Holds references (to the DATA_ vectors and beta_linkage_eff), so building one copies
93 * nothing. Stratum ids are 1-based, with 0 meaning "every level".
94 */
95template<class Type>
96struct LinkageTable {
97 const vector<int>& process; // process code (RCEATTLE_PROC_*)
98 const vector<int>& param; // parameter code within the process (codes above)
99 const vector<int>& species; // species id
100 const vector<int>& sex; // sex id
101 const vector<int>& age_bin; // age index (age - minage + 1), not the age itself
102 const vector<int>& fleet; // Fleet_code
103 const vector<int>& X_col; // 0-based column of X
104 const vector<int>& link; // RCEATTLE_LINK_* (note below)
105 const matrix<Type>& X; // design matrix [model year, row 0 = styr; column]
106 const vector<Type>& beta; // the row's estimated beta, or its random-effect value (5.2b)
107};

110// ---------------------------------------------------------------------
111// The five functions below share one pattern: for each row of `lt` naming this
112// process whose link matches `link_code`, add beta * X(yr, X_col) to the offset
113// for every stratum and year the row covers. `link_code` picks the rows:
114// RCEATTLE_LINK_IDENTITY: beta * x is added to the parameter's natural-scale value
115// (to the stored logit or log for DoubleNormalSS3 P2-P6, DoubleNormal's
116// logit floor and LogisticPM's log first-bin selectivity);
117// RCEATTLE_LINK_LOG: that value is multiplied by exp(beta * x);
118// RCEATTLE_LINK_LOGIT: reserved, refused in R;
119// RCEATTLE_LINK_EXPONENTIAL, catchability only: (log q + any log-link offset) is
120// multiplied by exp(beta * x), SS3's environmental link type 1.
121// Years beyond X keep an offset of 0, so env_data need not cover the
122// projection. See vignette("environmental-linkages-and-priors").
123// ---------------------------------------------------------------------

126/**
127 * @brief Add growth offsets to `growth_offset[sp, sex, yr, param]`.
128 *
129 * Growth is not age-specific, so the age-bin id is ignored. The SD codes
130 * (sd_L1 = 4, sd_Linf = 5) take intercept rows only and are skipped here.
131 *
132 * @param growth_offset [in,out] Offsets [nspp, max_sex, nyrs, n_growth_params].
133 * @param link_code,lt See the note above.
134 */
135template<class Type>
136void rceattle_apply_growth_linkages(
137 array<Type>& growth_offset,
138 int link_code,
139 const LinkageTable<Type>& lt,
140 int nspp,
141 const vector<int>& nsex,
142 int nyrs)
143{
144 int n = lt.beta.size();
145 if (n == 0) return;
146
147 int yr_hi = std::min(nyrs, (int)lt.X.rows());
148
149 for (int i = 0; i < n; ++i) {
150 if (lt.process(i) != RCEATTLE_PROC_GROWTH) continue;
151 if (lt.link(i) != link_code) continue;
152
153 int param = lt.param(i);
154 if (param < 0 || param >= RCEATTLE_N_GROWTH_PARAMS) continue;
155
156 int xc = lt.X_col(i);
157 Type b = lt.beta(i);
158
159 int sp_lo, sp_hi;
160 rceattle_stratum_range(lt.species(i), nspp, sp_lo, sp_hi);
161
162 for (int sp = sp_lo; sp < sp_hi; ++sp) {
163 int sx_lo, sx_hi;
164 rceattle_stratum_range(lt.sex(i), nsex(sp), sx_lo, sx_hi);
165
166 for (int sx = sx_lo; sx < sx_hi; ++sx) {
167 for (int yr = 0; yr < yr_hi; ++yr) {
168 growth_offset(sp, sx, yr, param) += b * lt.X(yr, xc);
169 }
170 }
171 }
172 }
173}

311/**
312 * @brief Add selectivity offsets, by fleet, sex and year, to one of five arrays.
313 *
314 * Offsets are per fleet; fleets sharing a `Selectivity_index` still share one
315 * parameter block through the map. The param code (RCEATTLE_SEL_*, above) picks
316 * the array: slopes `slp_offset`, inflections `inf_offset`, non-parametric
317 * coefficients `coff_offset` (every bin), apical height `apical_offset`, and
318 * DoubleNormalSS3 P1-P6 (SS3 pattern 24) `dn6_offset`.
319 *
320 * @param slp_offset [in,out] Slope offsets [2, n_flt, max_sex, nyrs].
321 * @param inf_offset [in,out] Inflection offsets [2, n_flt, max_sex, nyrs].
322 * @param coff_offset [in,out] Nonparametric offsets [n_flt, max_sex, n_sel_bins, nyrs].
323 * @param apical_offset [in,out] Apical-height offsets [n_flt, max_sex, nyrs].
324 * @param dn6_offset [in,out] DoubleNormalSS3 offsets [6, n_flt, max_sex, nyrs].
325 * @param link_code,lt See the note above.
326 */
327template<class Type>
328void rceattle_apply_sel_linkages(
329 array<Type>& slp_offset, // [2, n_flt, max_sex, nyrs]
330 array<Type>& inf_offset, // [2, n_flt, max_sex, nyrs]
331 array<Type>& coff_offset, // [n_flt, max_sex, n_sel_bins, nyrs]
332 array<Type>& apical_offset,// [n_flt, max_sex, nyrs]
333 array<Type>& dn6_offset, // [6, n_flt, max_sex, nyrs]
334 int link_code,
335 const LinkageTable<Type>& lt,
336 int n_flt,
337 int max_sex,
338 int n_sel_bins,
339 int nyrs)
340{
341 int n = lt.beta.size();
342 if (n == 0) return;
343
344 int yr_hi = std::min(nyrs, (int)lt.X.rows());
345
346 for (int i = 0; i < n; ++i) {
347 if (lt.process(i) != RCEATTLE_PROC_SEL) continue;
348 if (lt.link(i) != link_code) continue;
349
350 int param = lt.param(i);
351 int xc = lt.X_col(i);
352 Type b = lt.beta(i);
353
354 int fl_lo, fl_hi, sx_lo, sx_hi;
355 rceattle_stratum_range(lt.fleet(i), n_flt, fl_lo, fl_hi);
356 rceattle_stratum_range(lt.sex(i), max_sex, sx_lo, sx_hi);
357
358 for (int flt = fl_lo; flt < fl_hi; ++flt) {
359 for (int sx = sx_lo; sx < sx_hi; ++sx) {
360 for (int yr = 0; yr < yr_hi; ++yr) {
361 Type v = b * lt.X(yr, xc);
362 if (param == RCEATTLE_SEL_SLP_ASC || param == RCEATTLE_SEL_SLP_DESC) {
363 slp_offset(param - RCEATTLE_SEL_SLP_ASC, flt, sx, yr) += v;
364 } else if (param == RCEATTLE_SEL_INF_ASC || param == RCEATTLE_SEL_INF_DESC) {
365 inf_offset(param - RCEATTLE_SEL_INF_ASC, flt, sx, yr) += v;
366 } else if (param == RCEATTLE_SEL_COFF) {
367 for (int bin = 0; bin < n_sel_bins; ++bin) {
368 coff_offset(flt, sx, bin, yr) += v;
369 }
370 } else if (param == RCEATTLE_SEL_APICAL) {
371 apical_offset(flt, sx, yr) += v;
372 } else if (param >= RCEATTLE_SEL_DN_PEAK && param <= RCEATTLE_SEL_END_LOGIT) {
373 dn6_offset(param - RCEATTLE_SEL_DN_PEAK, flt, sx, yr) += v; // P1-P6
374 }
375 }
376 }
377 }
378 }

381/**
382 * @brief The random-effect slots of one group, in slot order, which is time order
383 * within a group (.assign_re_registry()).
384 *
385 * @param group_of_slot Each slot's group (ceattle.cpp's linkage_re_sigma: an index, not an sd).
386 * @param grp The group.
387 * @return The group's slot indices, ascending; empty if it has none.
388 */
389inline vector<int> rceattle_group_slots(const vector<int>& group_of_slot, int grp) {
390 int len = 0;
391 for (int s = 0; s < group_of_slot.size(); ++s) if (group_of_slot(s) == grp) len++;
392 vector<int> slots(len);
393 int j = 0;
394 for (int s = 0; s < group_of_slot.size(); ++s) if (group_of_slot(s) == grp) slots(j++) = s;
395 return slots;
396}

399/**
400 * @brief Add the negative log density of one group's deviations to two accumulators.
401 *
402 * RCEATTLE_RE_IID: sd sigma. RCEATTLE_RE_RW: sd sigma on first differences (the first
403 * deviation has no term of its own; build_map() holds it at its start, 0 by default,
404 * so the base parameter carries the level). RCEATTLE_RE_AR1: stationary, MARGINAL sd
405 * sigma and correlation rho (the glmmTMB / Rogers QAR1 convention).
406 *
407 * @param structure RCEATTLE_RE_IID, RCEATTLE_RE_RW or RCEATTLE_RE_AR1.
408 * @param re The group's deviations, in time order.
409 * @param sigma The group's sd, natural scale (AR1: the marginal sd).
410 * @param rho The AR1 correlation in (-1, 1); read only under RCEATTLE_RE_AR1.
411 * @param nll,nll_unweighted Cells of jnll_comp and unweighted_jnll_comp to add to.
412 */
413template<class Type>
414void rceattle_re_group_nll(int structure, vector<Type> re, Type sigma, Type rho,
415 Type& nll, Type& nll_unweighted)
416{
417 if (structure == RCEATTLE_RE_RW) {
418 for (int t = 1; t < re.size(); ++t) {
419 Type log_dens = dnorm(re(t) - re(t - 1), Type(0), sigma, true);
420 nll -= log_dens;
421 nll_unweighted -= log_dens;
422 }
423 } else if (structure == RCEATTLE_RE_AR1) {
424 Type neg_log_dens = density::SCALE(density::AR1(rho), sigma)(re); // a negative log density
425 nll += neg_log_dens;
426 nll_unweighted += neg_log_dens;
427 } else { // RCEATTLE_RE_IID
428 Type log_dens = sum(dnorm(re, Type(0), sigma, true));
429 nll -= log_dens;
430 nll_unweighted -= log_dens;
431 }
432}
