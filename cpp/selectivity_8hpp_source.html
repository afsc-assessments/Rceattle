4/**
5 * @brief Performs final normalization and projection of fishery selectivity (age/length) across all fleets.
6 *
7 * @details This function iterates through all fleets to process selectivity data.
8 * It handles:
9 * 1. Zeroing out selectivity for ages below bin_first_selected.
10 * 2. Normalizing selectivity values based on a single age (sel_norm_bin1 >= 0),
11 * an age range average (sel_norm_bin2 >= 0), or the maximum value
12 * across all ages/sexes (sel_norm_bin1 between -1 and -499).
13 * 3. Projecting the final year of hindcast selectivity into future projection years.
14 *
15 * @param flt fleet to iterate through.
16 * @param nyrs_hind Total number of years in the hindcast period.
17 * @param nyrs Total number of years including projection.
18 * @param flt_spp Function/Array mapping fleet index to species index.
19 * @param flt_sel_type Function/Array mapping fleet index to selectivity type.
20 * @param flt_sel_dim Age or length based selectivity.
21 * @param bin_first_selected Array/function returning the minimum age/length bin of selection for a fleet.
22 * @param nages Array/function returning the number of ages for a species.
23 * @param nlengths_pop Array/function returning the number of POPULATION length
24 * bins for a species, the grid a length-based curve is built on.
25 * @param nsex Array/function returning the number of sexes for a species.
26 * @param sel_norm_bin1 Array/function returning the normalization age or control flag.
27 * @param sel_norm_bin2 Array/function returning the upper bound for age-range normalization.
28 * @param sel_norm_scope Per fleet: 0 = WithinSex (each sex divided by its own
29 * reference, so both reach 1), 1 = AcrossSexes (one reference pooled over the
30 * sexes, so the less-selected sex stays below 1). Orthogonal to
31 * sel_norm_bin1/2, which say WHERE the reference is taken.
32 * @param selectivity 4D container (fleet, sex, age or length bin, year) modified in-place.
33 */
34template<class Type>
35void normalize_and_project_selectivity(
36 const int& flt,
37 const int& nyrs_hind,
38 const int& nyrs,

52 int sel_type = flt_sel_type(flt);
53 // A length-based curve lives on the POPULATION bins, so the zeroing,
54 // normalization and projection below all run over that grid.
55 int nbins = (flt_sel_dim(flt) == 0) ? nages(sp) : nlengths_pop(sp);
56
57 // Ages not selected
58 for(int yr = 0; yr < nyrs_hind; yr++) {
59 for(int bin = 0; bin < nbins; bin++){
60 for(int sex = 0; sex < nsex(sp); sex++){
61 if(bin < bin_first_selected(flt)){
62 selectivity(flt, sex, bin, yr) = 0;
63 }
64 }
65 }
66 }

68 // Normalization makes two orthogonal choices, carried in two inputs rather than
69 // encoded in the sign of one:
70 // sel_norm_bin1/2 -- WHERE the reference is taken: a named bin, the mean over
71 // a bin range, or (< 0) the maximum over bins;
72 // sel_norm_scope -- WHOSE scale it sets: WithinSex (0) gives each sex its own
73 // reference so both reach 1; AcrossSexes (1) pools one reference so the
74 // less-selected sex stays below 1 and relative sex selectivity survives.
75 // Identical for a one-sex species. Hake (5/12) normalizes in its own year/sex
76 // block above; LogisticPM (11) reuses sel_norm_bin1/2 as a penalty age-range.
77 // DoubleNormalSS3 (15) is SS3's unnormalized pattern 24: its plateau is 1 by
78 // construction and F multiplies the curve as it stands.
79 if((sel_norm_bin1(flt) > -500) && (sel_type != 5) && (sel_type != 12) && (sel_type != 11) && (sel_type != 15)) {
80 bool at_bin = (sel_norm_bin1(flt) >= 0);
81 bool over_range = at_bin && (sel_norm_bin2(flt) >= 0);
82 bool across = (sel_norm_scope(flt) == 1);
83
84 vector<Type> ref(nsex(sp));
85
86 for(int yr = 0; yr < nyrs_hind; yr++) {
87
88 // Reference value for each sex.
89 for(int sex = 0; sex < nsex(sp); sex++){
90 if(over_range){
91 // Mean selectivity over the bin range
92 ref(sex) = 0;
93 for(int bin = sel_norm_bin1(flt); bin <= sel_norm_bin2(flt); bin++) {
94 ref(sex) += selectivity(flt, sex, bin, yr)/(sel_norm_bin2(flt) - sel_norm_bin1(flt) + 1);
95 }
96 } else if(at_bin){
97 // Single normalization bin, floored so the division stays finite
98 ref(sex) = max2(Type(0.001), selectivity(flt, sex, sel_norm_bin1(flt), yr));
99 } else {
100 // Maximum over bins. Fold rather than branch: TMB tapes once, so an
101 // `if` on an AD Type would freeze the argmax bin at its initial
102 // position.
103 ref(sex) = 0;
104 for(int bin = 0; bin < nbins; bin++){
105 ref(sex) = max2(ref(sex), selectivity(flt, sex, bin, yr));
106 }
107 }
108 }
109
110 // One divisor for every sex, so their relative levels survive.
111 if(across){
112 Type pooled = ref(0);
113 for(int sex = 1; sex < nsex(sp); sex++) pooled = max2(pooled, ref(sex));
114 for(int sex = 0; sex < nsex(sp); sex++) ref(sex) = pooled;
115 }
116
117 for(int bin = 0; bin < nbins; bin++){
118 for(int sex = 0; sex < nsex(sp); sex++){
119 selectivity(flt, sex, bin, yr) /= ref(sex);
120 }
121 }
122 }
123 }
124
125 // Project outwards assuming final years selectivity
126 for(int yr = nyrs_hind; yr < nyrs; yr++) {
127 for(int bin = 0; bin < nbins; bin++){
128 for(int sex = 0; sex < nsex(sp); sex++){
129 selectivity(flt, sex, bin, yr) = selectivity(flt, sex, bin, nyrs_hind - 1);
130 }
131 }
132 }
133}

137 * @brief Converts length-based selectivity to age-based selectivity and stores it in a 4D array.
138 * * @details This function maps selectivity from length bins to age classes by
139 * integrating over a growth transition matrix (Prob(Length | Age)). The resulting
140 * age-based selectivity is stored directly into the provided 'sel_at_age' array for the
141 * specified fleet, sex, and year using pass by reference.
142 *
143 * @tparam Type The numeric type (e.g., double, TMB::adouble).
144 * @param flt Index of the fleet/fishery.
145 * @param sp Index of the species.
146 * @param sex Index of the sex.
147 * @param yr Index of the year.
148 * @param wtind Index for the weight-at-age/growth transition matrix type.
149 * @param nages Reference to vector containing number of ages per species.
150 * @param nlengths_pop Reference to vector containing number of POPULATION length bins per species.
151 * @param sel_length Reference to 4D array of selectivity-at-length [flt, sex, pop bin, yr].
152 * @param growth_matrix Reference to 5D transition matrix [wtind, sex, age, pop bin, yr].
153 * @param sel_at_age Reference to 4D output array [flt, sex, age, yr] to be updated in-place.
154 */
155template <class Type>
156void convert_length_selectivity(
157 const int& flt,
158 const int& sp,
159 const int& sex,
160 const int& yr,
161 const int& wtind,
162 const vector<int>& nages,
163 const vector<int>& nlengths_pop,
164 array<Type>& sel_length,
165 array<Type>& growth_matrix,
166 array<Type>& sel_at_age // Modified by reference
167) {
168 // Iterate through ages for the specific species
169 for(int age = 0; age < nages(sp); age++) {
170
171 // Initialize the specific cell in the 4D array to zero before accumulating
172 sel_at_age(flt, sex, age, yr) = 0.0;
173
174 // Summed over POPULATION bins: both arrays live on that grid, so the
175 // product Prob(Length|Age) * sel(Length) is formed at the resolution the
176 // curve is defined on rather than on a bin average.
177 for(int lp = 0; lp < nlengths_pop(sp); lp++) {
178 sel_at_age(flt, sex, age, yr) += growth_matrix(wtind, sex, age, lp, yr) * sel_length(flt, sex, lp, yr);
179 }
180 }
181}

184/**
185 * @brief Unified function to calculate selectivity across fleets, sexes, bins (age or length), and years.
186 * * @details This function calculates selectivity for the hindcast period,
187 * normalizes it, projects it into the future, and—if the base calculation was length-based—uses a
188 * growth transition matrix to output the equivalent age-based selectivity.
189 * * @section functional_forms Selectivity Models:
190 * The dimension (age vs length) is determined by `flt_sel_dim` (0 = age, 1 = length) and is shared
191 * across all parametric cases; the age/length distinction is handled inside each case, not by a
192 * separate case number. After calculation, length-based selectivity is converted to age-based via
193 * `convert_length_selectivity()`, and all selectivities are normalized/projected via
194 * `normalize_and_project_selectivity()`.
195 *
196 * - Case 0: Empirical — fixed selectivity read from `emp_sel` input; assumed age-based.
197 *
198 * - Case 1: Logistic — 2-parameter ascending logistic [Age or Length]
199 * sel(x) = 1 / (1 + exp(-slope * (x - inflection)))
200 * Parameters: log_sel_slp(0) = log(slope); sel_inf(0) = inflection point
201 *
202 * - Case 2: Non-parametric (Ianelli/AMAK style) [Age or Length]
203 * Bin-specific log-scale coefficients, normalized by mean; penalized for monotonicity and curvature.
204 * Parameters: sel_coff[flt, sex, bin] for bins 0..N_sel_bins-1
205 *
206 * - Case 3: Double Logistic (Dorn and Methot 1990) [Age or Length]
207 * sel(x) = asc(x) * (1 - desc(x)), where each limb is a 2-parameter logistic.
208 * Parameters: log_sel_slp(0/1) = log(ascending/descending slope); sel_inf(0/1) = ascending/descending inflection
209 *
210 * - Case 4: Descending Logistic [Age or Length]
211 * sel(x) = 1 - 1/(1 + exp(-slope * (x - inflection)))
212 * Parameters: log_sel_slp(1) = log(slope); sel_inf(1) = inflection point
213 *
214 * - Case 5: Non-parametric (Hake/Taylor style) [Age or Length]
215 * Cumulative sum of bin coefficients, exponentiated and normalized to max = 1.
216 * Parameters: sel_coff[flt, sex, bin] for bins bin_first_selected..N_sel_bins-1
217 *
218 * - Case 6: 2D AR1 (age x year) [Age or Length]
219 * Logistic-transformed bin-year deviates with AR1 correlation across age and year.
220 *
221 * - Case 7: 3D AR1 (age x cohort x year, Cheng et al 2024) [Age or Length]
222 * Conditional-variance parameterisation of a 3D random field across age, cohort and year.
223 *
224 * - Case 8: Double Normal [Age or Length]
225 * Four-parameter dome with a configurable right-tail floor (analogous to SS3 pattern
226 * 24 / P6). Gaussian ascending limb left of peak; Gaussian descending limb right of
227 * peak that approaches `right_floor` instead of 0; blended smoothly via a steep
228 * logistic weight so the function is twice differentiable everywhere.
229 * w(x) = 1 / (1 + exp(-20*(x - peak))) [~0 left of peak, ~1 right]
230 * asc(x) = exp(-0.5*((x-peak)/sigma_asc)^2)
231 * desc(x) = right_floor + (1 - right_floor) * exp(-0.5*((x-peak)/sigma_desc)^2)
232 * sel(x) = (1 - w(x)) * asc(x) + w(x) * desc(x)
233 * right_floor = 1 / (1 + exp(-(sel_inf(1) + sel_inf_dev(1)))) [logit]
234 * Parameters (reuses existing arrays; all four support annual deviates):
235 * sel_inf(0) = peak bin (mode); TV deviate: sel_inf_dev(0)
236 * log_sel_slp(0) = log(sigma_ascending); TV deviate: log_sel_slp_dev(0)
237 * log_sel_slp(1) = log(sigma_descending); TV deviate: log_sel_slp_dev(1)
238 * sel_inf(1) = logit(right_floor) (~SS3 P6/end_logit). logit -> -Inf gives a
239 * fully dome-shaped curve; logit -> +Inf collapses to logistic
240 * (ascending only). TV deviate: sel_inf_dev(1)
241 *
242 * - Case 15: DoubleNormalSS3 [Age or Length]
243 * Stock Synthesis size-selectivity pattern 24 (and age pattern 20), evaluated at the
244 * bin midpoints x with SS3's own six parameters, each on SS3's scale:
245 * P1 peak (cm or age), P2 top width (logit), P3 ascending width (log),
246 * P4 descending width (log), P5 initial selectivity (logit), P6 final (logit).
247 * peak2 = P1 + w + (0.99 x_last - P1 - w) / (1 + exp(-P2)), w = bin width
248 * asc = p1 + (1 - p1)(exp(-(x-P1)^2/e^P3) - t1min)/(1 - t1min), p1 = logistic(P5)
249 * dsc = 1 + (p2 - 1)(exp(-(x-peak2)^2/e^P4) - 1)/(t2min - 1), p2 = logistic(P6)
250 * j(t) = 1/(1 + exp(-20 t/(1 + |t|)))
251 * sel = asc (1 - j(x-P1)) + j(x-P1) ((1 - j(x-peak2)) + dsc j(x-peak2))
252 * t1min / t2min are the Gaussian limbs at the first and last bins. A P5 or P6 of
253 * -999 or -1000 in SS3 drops that end's scaling (asc = exp(-(x-P1)^2/e^P3), likewise
254 * dsc); here that choice is the data flag sel_dn6_ends. No normalization: the curve
255 * peaks at 1 by construction. Parameters live in their own array, sel_dn6[6, flt, sex];
256 * time variation is through linkages on each of the six, value = (P + identity
257 * offset) x exp(log offset), so an SS3 block replacement (identity link, block
258 * factor) followed by multiplicative annual devs (log link) reproduces SS3's order.
259 * * @param nspp Number of species.
260 * @param n_flt Number of fleets (fisheries and surveys).
261 * @param nyrs Total years (including hindcast and projection).
262 * @param nyrs_hind Number of hindcast years.
263 * @param styr Start year (for indexing empirical data).
264 * @param nsex Vector of number of sexes per species.
265 * @param nages Vector of max age bins per species.
266 * @param nlengths Vector of max length bins per species.
267 * @param lengths Matrix of length bin boundaries (nspp x max_length_bins).
268 * @param flt_spp Vector mapping fleet index to species index.
269 * @param flt_sel_type Vector of selectivity form.
270 * @param flt_sel_dim Age or length based selectivity
271 * @param bin_first_selected Vector of minimum age/length bins selected per fleet.
272 * @param flt_n_sel_bins Vector of maximum bins with estimated coefficients per fleet.
273 * @param sel_norm_bin1 Vector of bins used for normalization (or control flags) per fleet.
274 * @param sel_norm_bin2 Vector of upper bins for mean normalization per fleet.
275 * @param emp_sel_obs Matrix of empirical selectivity observation values.
276 * @param emp_sel_ctl Control matrix for empirical selectivity indexing.
277 * @param log_sel_slp Matrix of log-scale logistic slope parameters.
278 * @param log_sel_slp_dev 4D array of yearly slope deviations.
279 * @param sel_inf Matrix of bin-at-inflection parameters.
280 * @param sel_inf_dev 4D array of yearly inflection deviations.
281 * @param sel_coff 3D array of bin-specific coefficients for non-parametric models.
282 * @param sel_coff_dev 4D array of yearly coefficient deviations.
283 * @param avg_sel [Output] 3D array of average selectivity per fleet/sex/year (modified by reference).
284 * @param non_par_sel [Output] 4D array of unnormalized non-parametric selectivity (modified by reference).
285 * @param log_non_par_sel [Output] The same curve on the log scale, which is the scale the
286 * non-parametric selectivity penalties are written on (modified by reference).
287 * @param sel_at_length [Output] 4D array of final length-based selectivity (modified by reference).
288 * @param sel_at_age [Output] 4D array of final age-based selectivity (modified by reference).
289 * @param growth_matrix 5D array defining the length-at-age probability transition matrix.
290 */

388 // base coefficients when its deviations are a walk, otherwise the shared
389 // non-parametric branch. Time_varying_sel picks which, so the switch below
390 // dispatches on this rather than on the Selectivity code alone.

398 // width; an age-based one over ages. Identical to the data grid whenever no
399 // population grid was supplied.
400 int nbins = is_length_based? nlengths_pop(sp) : nages(sp);
401 int n_sel_bins = flt_n_sel_bins(flt);
402 // Bin midpoints, SS3's len_bins_m. A population length grid need not be
403 // uniform -- SS3 models routinely widen the tail bins -- so each bin takes
404 // its OWN width rather than the first bin's, which otherwise evaluated every
405 // curve away from the bin it labels and disagreed with the age-length key it
406 // multiplies. Identical on a uniform grid, so no existing model moves.
407 // A single length bin has no neighbour to take a width from, so it keeps the

422 // bin's where the grid is not uniform (SS_readdata_330.tpl:1644). Only the
423 // DoubleNormalSS3 peak2 reads it, exactly as SS_selex.tpl:153 does.

425 ? (lengths_pop(sp, nbins / 2) - lengths_pop(sp, nbins / 2 - 1)) : Type(1.0);

428 // NonParametricPM (type 9) random walk (the realized curve is then capped).

438 // slope stored on the log scale: log-link offset rides inside the
439 // exp (multiplicative on the natural slope), natural-link offset
440 // adds after. inflection stored natural: identity-link offset adds,
441 // log-link offset multiplies. Both no-op when the offsets are zero.
442 Type slope = exp(log_sel_slp(0, flt, sex) + log_sel_slp_dev(0, flt, sex, yr)
443 + sel_slp_off(0, flt, sex, yr)) + sel_slp_off_nat(0, flt, sex, yr);

447
448 if (is_length_based) sel_at_length(flt, sex, bin, yr) = val;
449 else sel_at_age(flt, sex, bin, yr) = val;

451 break;

453 // NonParametricIntegrable (13) under Time_varying_sel = "RandomWalk": the
454 // Ianelli curve on the base coefficients plus the running sum of increments
455 // (years after the fleet's start year), centred for output only, so with no
456 // increments every year is NonParametric's curve. No cap; the increments are
457 // scored in ceattle.cpp. Under "Off" or "IID" the form falls to the shared
458 // non-parametric branch below.
