5/**
6 * @brief Transforms predator-prey preference parameters into vulnerability.
7 *
8 * Uses a multinomial logistic transformation to ensure that the sum of
9 * predator-prey preferences and the vulnerability to "other food" sums to 1.
10 * This is only applied to species using parametric suitability (suitMode > 0).
11 * Adopted from Trijoulet et al. 2020 (https://github.com/vtrijoulet/Multisp_model_JAE/blob/master/MS_SSM.cpp).
12 *
13 * @param vulnerability Matrix updated with bounded predator-prey vulnerability [nspp, nspp].
14 * @param vulnerability_other Vector updated with the remaining vulnerability for "other food" [nspp].
15 * @param nspp Total number of species in the model.
16 * @param suitMode Vector specifying the suitability/preference mode for each species.
17 * @param log_phi Matrix of estimated log-preference parameters [nspp, nspp].
18 */
19template <class Type>
20void calculate_vulnerability(matrix<Type> &vulnerability,
21 vector<Type> &vulnerability_other,
22 int nspp,
23 const vector<int> &suitMode,
24 matrix<Type> &log_phi) {
25
26 // Reset arrays to ensure clean calculations
27 vulnerability.setZero();
28 vulnerability_other.setZero();
29
30 for(int rsp = 0; rsp < nspp; rsp++) {
31 if(suitMode(rsp) > 0) {
32
33 Type sum_phi = 0; // Sum of predator-prey preference coefficients for multinomial transformation
34
35 // 1. Calculate the sum of exponentiated preferences
36 for(int ksp = 0; ksp < nspp; ksp++) {
37 sum_phi += exp(log_phi(rsp, ksp));
38 }
39
40 // 2. Apply multinomial logistic transformation
41 for(int ksp = 0; ksp < nspp; ksp++) {
42 vulnerability(rsp, ksp) = exp(log_phi(rsp, ksp)) / (Type(1.0) + sum_phi);
43 }
44
45 // 3. Assign the remainder to "other food"
46 vulnerability_other(rsp) = Type(1.0) - vulnerability.row(rsp).sum();
47 }
48 }
49}

51/**
52 * @brief Calculates parametric suitability (Gamma, Lognormal, and Normal forms) based on predator/prey size ratios.
53 *
54 * This function merges length-based and weight-based suitability calculations into a single,
55 * highly optimized pass. It calculates the relative size ratio of predators to prey and evaluates
56 * that ratio against a specified probability distribution function to determine suitability.
57 * The `suitMode` switch dictates both the distribution and the biological metric used:
58 * - suitMode = 1: Gamma distribution (Length-based) [NOT YET AVAILABLE -- blocked in data_check()]
59 * - suitMode = 2: Gamma distribution (Weight-based)
60 * - suitMode = 3: Lognormal distribution (Length-based) [NOT YET AVAILABLE -- blocked in data_check()]
61 * - suitMode = 4: Lognormal distribution (Weight-based)
62 * - suitMode = 5: Normal distribution (Length-based) [NOT YET AVAILABLE -- blocked in data_check()]
63 * - suitMode = 6: Normal distribution (Weight-based)
64 * Note: length-based modes (1, 3, 5) require a growth model to be active and are blocked
65 * at the R level until a growth-model validation pass is added to data_check().
66 *
67 * @param suitability Array updated with size-based suitability [nspp*max_sex, nspp*max_sex, max_age, max_age, nyrs].
68 * @param suit_other Array updated with the baseline vulnerability to "other food" [nspp, max_sex, max_age, nyrs].
69 * @param nspp Total number of species in the model.
70 * @param nyrs Total number of years evaluated.
71 * @param nsex Vector containing the number of sexes modeled for each species.
72 * @param nages Vector containing the number of age classes for each species.
73 * @param suitMode Vector specifying the parametric suitability mode for each predator species.
74 * @param length_hat Array of estimated lengths-at-age [nspp*2+n_flt, max_sex, max_age, nyrs].
75 * @param weight_hat Array of estimated weights-at-age [nspp*2+n_flt, max_sex, max_age, nyrs].
76 * @param vulnerability Matrix of bounded predator-prey vulnerability preferences [nspp, nspp].
77 * @param vulnerability_other Vector of vulnerability preference for "other food" [nspp].
78 * @param gam_a Vector of distribution shape/mean parameters for predator size-selectivity [nspp].
79 * @param gam_b Vector of distribution scale/sd parameters for predator size-selectivity [nspp].
80 */
81template <class Type>
82void calculate_parametric_suitability(array<Type> &suitability,
83 array<Type> &suit_other,
84 int nspp, int nyrs,
85 const vector<int> &nsex,
86 const vector<int> &nages,
87 const vector<int> &suitMode,
88 array<Type> &length_hat,
89 array<Type> &weight_hat,
90 matrix<Type> &vulnerability,
91 const vector<Type> &vulnerability_other,
92 const vector<Type> &gam_a,
93 const vector<Type> &gam_b) {
94
95 for(int rsp = 0; rsp < nspp; rsp++) {
96 int smode = suitMode(rsp);
97 int wt_idx_rsp = 2 * rsp;
98
99 // Parametric suitability modes (1 through 6)
100 if(smode >= 1 && smode <= 6) {
101 for(int r_sex = 0; r_sex < nsex(rsp); r_sex++) {
102 int r_idx = rsp + (nspp * r_sex);
103
104 for(int r_age = 0; r_age < nages(rsp); r_age++) {
105 for(int ksp = 0; ksp < nspp; ksp++) {
106 int wt_idx_ksp = 2 * ksp;
107
108 for(int k_sex = 0; k_sex < nsex(ksp); k_sex++) {
109 int k_idx = ksp + (nspp * k_sex);
110
111 for(int k_age = 0; k_age < nages(ksp); k_age++) {
112 for(int yr = 0; yr < nyrs; yr++) {
113
114 suit_other(rsp, r_sex, r_age, yr) = vulnerability_other(rsp);
115 Type size_ratio = 0.0;
116
117 // 1. Calculate the size ratio metric based on suitMode
118 // - suitMode 1/2 = Gamma (Length/Weight)
119 // - suitMode 3/4 = Lognormal (Length/Weight)
120 // - suitMode 5/6 = Normal (Length/Weight)
121 if(smode == 1 || smode == 3) {
122 size_ratio = log(length_hat(rsp, r_sex, r_age, yr) / length_hat(ksp, k_sex, k_age, yr));
123 } else if (smode == 2 || smode == 4) {
124 size_ratio = log(weight_hat(wt_idx_rsp, r_sex, r_age, yr) / weight_hat(wt_idx_ksp, k_sex, k_age, yr)); // Log ratio of weights
125 } else if (smode == 5) {
126 size_ratio = length_hat(rsp, r_sex, r_age, yr) / length_hat(ksp, k_sex, k_age, yr);
127 } else if (smode == 6) {
128 size_ratio = weight_hat(wt_idx_rsp, r_sex, r_age, yr) / weight_hat(wt_idx_ksp, k_sex, k_age, yr);
129 }
130
131 // 2. Apply the distribution density function
132 if(size_ratio > 0) {
133 if(smode == 1 || smode == 2) {
134 // Gamma distribution
135 suitability(r_idx, k_idx, r_age, k_age, yr) = vulnerability(rsp, ksp) * dgamma(size_ratio, gam_a(rsp), gam_b(rsp)) /
136 dgamma((gam_a(rsp)-1) * gam_b(rsp), gam_a(rsp), gam_b(rsp)); // Scale to 0,1 by dividing by max
137 } else if (smode >= 3 && smode <= 6) {
138 // Normal/Lognormal distribution
139 suitability(r_idx, k_idx, r_age, k_age, yr) = vulnerability(rsp, ksp) * dnorm(size_ratio, gam_a(rsp), gam_b(rsp)) /
140 dnorm(gam_a(rsp), gam_a(rsp), gam_b(rsp)); // Scale to 0,1 by dividing by max
141 }
142 }
143 }
144 }
145 }
146 }
147 }

155 * @brief Calculates MSVPA/empirical-based suitability based on stomach content and prey biomass.
156 *
157 * This function calculates suitability for predator species where suitMode == 0.
158 * It first computes the ratio of stomach proportion to available prey biomass (stom_div_bio).
159 * It then normalizes this ratio across all included prey and "other food" to determine suitability.
160 * The suitability is averaged over a specified reference period (suit_styr to suit_endyr)
161 * and held constant across all years. Two working arrays are allocated inside:
162 * suma_suit, the sum of stom_div_bio over prey, and diet_prop_sum, the sum of
163 * diet proportions, each [nspp, max_sex, max_age, nyrs].
164 *
165 * @param stom_div_bio Array updated with the ratio of diet proportion to biomass [nspp*max_sex, nspp*max_sex, max_age, max_age, nyrs].
166 * @param suitability Array updated with the final calculated suitability [nspp*max_sex, nspp*max_sex, max_age, max_age, nyrs].
167 * @param suit_other Array updated with the remaining suitability for "other food" [nspp, max_sex, max_age, nyrs].
168 * @param nspp Total number of species in the model.
169 * @param nyrs Total number of years evaluated.
170 * @param suit_styr Start year index (per predator species) for the suitability averaging period.
171 * @param suit_endyr End year index (per predator species) for the suitability averaging period.
172 * @param nyrs_suit Number of years in the suitability averaging period, per predator species.
173 * @param msmMode Integer specifying the multi-species predation mode (e.g., 2 = Type III MSVPA).
174 * @param nsex Vector containing the number of sexes modeled for each species.
175 * @param nages Vector containing the number of age classes for each species.
176 * @param suitMode Vector specifying the suitability mode for each predator species.
177 * @param diet_prop Array of observed or estimated diet proportions.
178 * @param avgN_at_age Array of average numbers-at-age.
179 * @param weight_hat Array of estimated weights-at-age.
180 * @param other_food_diet_prop Array of estimated diet proportion belonging to "other food".
181 */

215 // ======================================================================

218 int wt_idx_ksp = 2 * ksp;
219 for(int k_sex = 0; k_sex < nsex(ksp); k_sex++) {
220 int k_idx = ksp + (nspp * k_sex);
221
222 for(int k_age = 0; k_age < nages(ksp); k_age++) {

224 // OPTIMIZATION 1: Calculate prey biomass denominator once per prey
225 Type prey_wt = weight_hat(wt_idx_ksp, k_sex, k_age, yr);

228
229 Type prey_denom = prey_wt * prey_N;

233 Type inv_prey_denom = Type(1.0) / prey_denom; // Convert to fast multiplication
234
235 for(int r_sex = 0; r_sex < nsex(rsp); r_sex++) {
236 int r_idx = rsp + (nspp * r_sex);
237 for(int r_age = 0; r_age < nages(rsp); r_age++) {
238
239 suitability(r_idx, k_idx, r_age, k_age, yr) = 0.0;
240 Type d_prop = diet_prop(r_idx, k_idx, r_age, k_age, yr);
241
242 // Only process if predator actually ate this prey
243 if (d_prop > Type(0.0)) {
244 Type sdb = d_prop * inv_prey_denom;
245 stom_div_bio(r_idx, k_idx, r_age, k_age, yr) = sdb;
246 suma_suit(rsp, r_sex, r_age, yr) += sdb;
247 diet_prop_sum(rsp, r_sex, r_age, yr) += d_prop;
248 }
249 }
250 }
251 }
252 }
253 }
254 }
255 }

259 // ======================================================================
260 for(int r_sex = 0; r_sex < nsex(rsp); r_sex++) {
261 int r_idx = rsp + (nspp * r_sex);
262 for(int r_age = 0; r_age < nages(rsp); r_age++) {

272 inv_pred_denom(yr) = Type(1.0) / max2(denom, Type(1.0e-10));
273 } else {
274 inv_pred_denom(yr) = Type(0.0);
275 }
276 }
277

279
280 for(int ksp = 0; ksp < nspp; ksp++) {
281 for(int k_sex = 0; k_sex < nsex(ksp); k_sex++) {
282 int k_idx = ksp + (nspp * k_sex);
283 for(int k_age = 0; k_age < nages(ksp); k_age++) {
284
285 Type suit_sum_yrs = Type(0.0);
286
287 // Accumulate standardized suitability over the reference years
288 for(int yr = suit_styr(rsp); yr <= suit_endyr(rsp); yr++) {
289 suit_sum_yrs += stom_div_bio(r_idx, k_idx, r_age, k_age, yr) * inv_pred_denom(yr);
290 // stom_div_bio(r_idx, k_idx, r_age, k_age, yr) / (suma_suit(rsp, r_sex, r_age, yr ) + other_food_diet_prop(rsp, r_sex, r_age, yr));
291 }
292
293 Type final_suit = suit_sum_yrs / Type(nyrs_suit(rsp));
294 total_suit_prey += final_suit;
295
296 // Fill all years
297 for(int yr = 0; yr < nyrs; yr++) {
298 suitability(r_idx, k_idx, r_age, k_age, yr) = final_suit;
299 }
300 }
301 }
302 }
303
304 // OPTIMIZATION 3: Assign "other food" suitability efficiently for all years
305 Type final_suit_other = Type(1.0) - total_suit_prey;
306 for(int yr = 0; yr < nyrs; yr++) {
307 suit_other(rsp, r_sex, r_age, yr) = final_suit_other;
308 }
309 }
310 }
311 }

316/**
317 * @brief Calculates available food, predation mortality (M2), and consumed biomass (MSVPA).
318 *
319 * This function calculates the core MSVPA predation dynamics in two passes.
320 * Pass 1 calculates the total available food for each predator. Pass 2 allocates
321 * predation mortality (M2) and consumption across prey species based on suitability
322 * and available food. Dynamic reference points (dB0 and dBF) are calculated simultaneously.
323 * * Highly optimized for TMB:
324 * 1. Prey loops are inverted in Step 1 to prevent redundant prey biomass calculations.
325 * 2. Predator-specific division scalars are hoisted out of the deepest loops in Step 2.
326 * 3. Year loops are moved to the outermost level to maximize column-major cache locality.
327 *
328 * @param avail_food Array updated with total available food [nspp, max_sex, max_age, nyrs].
329 * @param avail_food_dB0 Array updated with available food at F=0.
330 * @param avail_food_dBF Array updated with available food at F=Ftarget.
331 * @param M2_at_age Array updated with calculated predation mortality [nspp, max_sex, max_age, nyrs].
332 * @param M2_at_age_dB0 Array updated with predation mortality at F=0.
333 * @param M2_at_age_dBF Array updated with predation mortality at F=Ftarget.
334 * @param M2_prop Array updated with relative predation mortality.
335 * @param B_eaten Array updated with biomass of prey eaten by specific predator.
336 * @param B_eaten_as_prey Array updated with total biomass of prey eaten by all predators.
337 * @param diet_prop_hat Array updated with predicted diet proportions.
338 * @param other_diet_prop_hat Array updated with predicted proportion of "other food" in diet.
339 * @param nspp Total number of species in the model.
340 * @param nyrs Total number of years evaluated.
341 * @param nsex Vector containing the number of sexes modeled for each species.
342 * @param nages Vector containing the number of age classes for each species.
343 * @param msmMode Integer specifying the multi-species predation mode (1 = Type II, 2 = Type III).
344 * @param avgN_at_age Array of average numbers-at-age.
345 * @param avgN_at_age_dB0 Array of average numbers-at-age at F=0.
346 * @param avgN_at_age_dBF Array of average numbers-at-age at F=Ftarget.
347 * @param suitability Array of calculated suitability [nspp*max_sex, nspp*max_sex, max_age, max_age, nyrs].
348 * @param suit_other Array of suitability for "other food".
349 * @param weight_hat Array of estimated weights-at-age.
350 * @param ration Array of calculated individual consumption/ration (kg/yr).
351 * @param other_food Vector of base biomass for "other food" by predator.

483 for(int k_sex = 0; k_sex < nsex(ksp); k_sex++) {

485 for(int k_age = 0; k_age < nages(ksp); k_age++) {
