4/**
5 * @brief Parses empirical diet observations and maps them into a 5D proportion array.
6 * * @details This function takes the raw diet observation matrix (`diet_obs`) and its
7 * corresponding control matrix (`diet_ctl`) and maps the observed proportions into
8 * the structured 5D `diet_prop` array for MSVPA based suitability.
9 * It dynamically handles indexing for 1-sex
10 * versus 2-sex models. If the control matrix specifies a year `flt_yr == 0`, the
11 * observed proportion is applied as an average across all modeled years. Data
12 * rows missing specific predator or prey ages are skipped during this step.
13 * * @param nspp Number of modeled species.
14 * @param nyrs Total number of years in the model (hindcast + projection).
15 * @param nyrs_hind Number of years in the hindcast period.
16 * @param styr The starting year of the model.
17 * @param minage Vector containing the minimum modeled age for each species.
18 * @param nsex Vector containing the number of sexes modeled for each species (1 or 2).
19 * @param diet_obs Matrix of empirical diet observations (column 1 contains the proportions).
20 * @param diet_ctl Control matrix mapping each observation row to predator, prey, sexes, ages, and year.
21 * @param diet_prop [Output] 5D array (pred_idx, prey_idx, pred_age, prey_age, yr) of observed diet proportions. Modified by reference.
22 */
23template <class Type>
24void organize_diet_obs(
25 const int& nspp,

27 const int& nyrs_hind,
28 const int& styr,
29 const vector<int>& minage,
30 const vector<int>& nsex,
31 matrix<Type>& diet_obs,
32 matrix<int>& diet_ctl,

34) {
35 for(int stom_ind = 0; stom_ind < diet_obs.rows(); stom_ind++) {
36 int rsp = diet_ctl(stom_ind, 0) - 1; // Index of pred
37 int ksp = diet_ctl(stom_ind, 1) - 1; // Index of prey
38 int r_sex = diet_ctl(stom_ind, 2); // Index of pred sex
39 int k_sex = diet_ctl(stom_ind, 3); // Index of prey sex
40 int r_age = diet_ctl(stom_ind, 4) - minage(rsp); // Index of pred age
41 int k_age = diet_ctl(stom_ind, 5) - minage(ksp); // Index of prey age
42 int flt_yr = diet_ctl(stom_ind, 6); // Index of year
43 // Diet proportion of prey-at-at in predator-at-age
44 // if K_age < 0, data are diet proportion of prey-spp in predator-at-age (summed across prey ages) and are not used for MSVPA based suitability
45 // if k_age < 0 & r_age < 0, data are diet proportion of prey-spp in predator-spp (summed across prey ages and averaged across predator ages)
46 // and are not used for MSVPA based suitability
47
48 // Only process if specific ages are provided for both predator and prey
49 if((k_age < 0) || (r_age < 0)) continue;
50
51 // Determine sex looping bounds for predator (0 = combined, 1 = female, 2 = male)
52 int r_start = 0;
53 int r_end = 0;
54 if(nsex(rsp) == 2) {
55 if(r_sex == 0) { r_end = 1; } // Loop over both sexes
56 else { r_start = r_sex - 1; r_end = r_sex - 1; } // Specific sex
57 }
58
59 // Determine sex looping bounds for prey
60 int k_start = 0;
61 int k_end = 0;
62 if(nsex(ksp) == 2) {
63 if(k_sex == 0) { k_end = 1; } // Loop over both sexes
64 else { k_start = k_sex - 1; k_end = k_sex - 1; } // Specific sex
65 }
66
67 // Value to assign
68 Type obs_val = diet_obs(stom_ind, 1);
69
70 // Assign Diet proportion
71 for(int j = r_start; j <= r_end; j++) {
72 int r_idx = rsp + (nspp * j);
73
74 for(int k = k_start; k <= k_end; k++) {
75 int k_idx = ksp + (nspp * k);
76
77 // Specific year data
78 if(flt_yr > 0) {
79 int yr = flt_yr - styr;
80 if(yr >= 0 && yr < nyrs_hind) {
81 diet_prop(r_idx, k_idx, r_age, k_age, yr) = obs_val;
82 }
83 }
84 // Average across all years
85 else if(flt_yr == 0) {
86 for(int yr = 0; yr < nyrs; yr++) {
87 diet_prop(r_idx, k_idx, r_age, k_age, yr) = obs_val;
88 }
89 }
90 }

139/**
140 * @brief Predicts the expected stomach content proportions given the modeled population state.
141 * * @details This function maps the model's internal prediction of diet proportions (`diet_prop_hat`)
142 * back to the structure of the original empirical data (`diet_obs`). Because empirical
143 * stomach data is often aggregated across ages or sexes, this function dynamically
144 * calculates the appropriate expected value using abundance-weighted averages (`avgN_at_age`).
145 * * It handles four types of data aggregation:
146 * - **Type 1:** Specific predator age and specific prey age (weighted across sexes if combined).
147 * - **Type 2:** Specific predator age, summed across all prey ages.
148 * - **Type 3:** Unweighted arithmetic mean across predator ages, summed across prey ages.
149 * - **Type 4:** Abundance-weighted mean across predator ages, summed across prey ages.
150 * * @param nspp Number of modeled species.
151 * @param nyrs_hind Number of years in the hindcast period.
152 * @param styr The starting year of the model.
153 * @param suit_styr The starting year (per predator species) used for suitability averaging.
154 * @param suit_endyr The ending year (per predator species) used for suitability averaging.
155 * @param minage Vector containing the minimum modeled age for each species.
156 * @param nsex Vector containing the number of sexes modeled for each species.
157 * @param nages Vector containing the maximum number of age bins for each species.
158 * @param diet_obs Matrix of empirical diet observations (used here for row-matching).
159 * @param diet_ctl Control matrix mapping each observation row to specific aggregation levels.
160 * @param avgN_at_age 4D array of average population numbers-at-age (used for weighting).
161 * @param diet_prop_hat 5D array of the model's predicted diet proportions for specific ages/sexes.
162 * @param diet_hat [Output] Matrix of predicted diet proportions formatted to match `diet_obs`. Modified by reference.
163 */

165void predict_stomach_content(
166 const int& nspp,

239 }

265 }
266 }

268 }

270 }
271
272 // TYPE 4: Weighted mean across pred ages, sum across prey ages
273 else if((k_age < 0) & (r_age < -500)) {
274 Type weighted_sum = 0;
275 Type total_numbers = 0;
276 for(int r_s = r_start; r_s <= r_end; r_s++) {
277 for(int r_age_tmp = 0; r_age_tmp < nages(rsp); r_age_tmp++) {
278 Type pred_numbers = avgN_at_age(rsp, r_s, r_age_tmp, yr);
279 total_numbers += pred_numbers;
280 for(int k_s = k_start; k_s <= k_end; k_s++) {
281 for(int k_age_tmp = 0; k_age_tmp < nages(ksp); k_age_tmp++) {
282 weighted_sum += diet_prop_hat(rsp + (nspp * r_s), ksp + (nspp * k_s), r_age_tmp, k_age_tmp, yr) * pred_numbers;
283 }
284 }
285 }
286 }
287 if(total_numbers > 0) type_accum += weighted_sum / total_numbers;
288 }
289 yr_accum += type_accum;
290 }
291 diet_hat(stom_ind, 1) = yr_accum / n_yrs_evaluated;
292 }
293 }
